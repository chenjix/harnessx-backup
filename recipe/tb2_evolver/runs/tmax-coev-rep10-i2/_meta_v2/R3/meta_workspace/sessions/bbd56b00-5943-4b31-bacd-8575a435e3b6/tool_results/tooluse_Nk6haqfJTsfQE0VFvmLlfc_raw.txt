3	"""FunctionalVerifyGateProcessor.
4	
5	A stateful exit gate that supersedes the single-shot self-verify nudge.
6	
7	Observed failure mode (R0): 23 of 29 failing tasks exited with
8	``exit_reason=done`` — the agent believed it was finished. The existing
9	single-shot ``CustomSelfVerifyProcessor`` fires exactly once; the agent
10	answers it with a *shallow static re-check* (``ls`` the output file,
11	``grep`` for a keyword, ``cat`` the script) and immediately re-exits.
12	The verifier then fails the task on a semantic / functional bug that a
13	real end-to-end run would have surfaced (wrong numeric constant, a
14	daemon that never actually keeps disk under quota, a query that uses the
15	wrong index, etc.).
16	
17	This processor keeps the good checklist content but makes the gate
18	*functional-aware*:
19	
20	- When the agent tries to exit, it is nudged to verify.
21	- After a nudge, the processor watches whether the agent runs a
22	  *substantive* command (something that actually executes / exercises the
23	  solution) as opposed to purely read-only inspection (``ls`` / ``cat`` /
24	  ``grep`` / ``head`` / ``find`` / ``stat`` / ``wc`` / ``echo``).
25	- If the agent tries to exit again having only done read-only inspection,
26	  it is nudged once more with an escalated, functional-testing demand.
27	- The gate never blocks more than ``max_nudges`` times, so a task that is
28	  genuinely done (or genuinely stuck) always terminates — no infinite
29	  loop, bounded extra token cost.
30	
31	Class of tasks served: every TB2 task whose correctness is checked by a
32	functional final-state test (i.e. essentially all of them). It carries no
33	task-specific literals.
34	"""
35	from __future__ import annotations
36	
37	import dataclasses
38	import re
39	import uuid
40	
41	from harnessx.core.events import (
42	    BeforeModelEvent,
43	    Message,
44	    ModelResponseEvent,
45	    TaskEndEvent,
46	    TaskStartEvent,
47	    ToolCall,
48	    ToolCallEvent,
49	)
50	from harnessx.core.processor import MultiHookProcessor
51	
52	_TOOL = "_tb2_functional_verify"
53	_ACK = "Verification check initiated. See the message above for instructions."
54	
55	# Commands that only *inspect* state — running one of these is not evidence
56	# that the agent actually exercised its solution end to end.
57	_READONLY_HEAD_RE = re.compile(
58	    r"^\s*(ls|cat|head|tail|grep|egrep|fgrep|find|stat|wc|echo|file|"
59	    r"pwd|which|type|env|printenv|tree|du|df|realpath|readlink|basename|dirname)\b"
60	)
61	
62	
63	def _is_substantive(command: str) -> bool:
64	    """True if the bash command plausibly *executes / exercises* the solution.
65	
66	    Heuristic: a command is substantive unless every top-level segment
67	    (split on ``;``, ``&&``, ``||`` and newlines) begins with a read-only
68	    inspection verb. This treats ``python run.py`` / ``make`` / ``./daemon &``
69	    / ``bash test.sh`` / ``curl localhost`` as substantive, while
70	    ``ls -lh out.txt && cat out.txt`` stays shallow.
71	    """
72	    if not command or not command.strip():
73	        return False
74	    segments = re.split(r"(?:;|&&|\|\||\n)+", command)
75	    for seg in segments:
76	        seg = seg.strip()
77	        if not seg:
78	            continue
79	        if not _READONLY_HEAD_RE.match(seg):
80	            return True
81	    # All segments were read-only inspection (or there were none).
82	    return False
83	
84	
85	_NUDGE_1 = """\
86	Before finishing, VERIFY YOUR WORK FUNCTIONALLY — do not skip this even if you already glanced at the files.
87	
88	1. **Re-read the task description now.** List every concrete requirement: exact output paths, exact numeric constants/thresholds, exact formats, accuracy criteria, and any runtime behavior (a service that must stay up, a monitor that must keep something bounded, a script that must produce a specific result).
89	
90	2. **Actually RUN your solution end to end against the real scenario.** Static checks are NOT verification: confirming a file exists (`ls`), that it contains a keyword (`grep`), or that a script parses is worthless — those pass even on a broken solution. Instead execute the real workflow: run the program with the real inputs, drive the actual scenario the task describes, and observe what happens.
91	
92	3. **Compare observed behavior to the required behavior**, value by value. If the task states an exact constant, threshold, or expected result, recompute it independently and check it matches — do not trust your own earlier output.
93	
94	4. **For services/daemons:** start them and confirm they are alive AND behave correctly under the described load right now.
95	
96	Run the real functional test now, then report the observed result. When — and only when — a genuine end-to-end run confirms correctness, end your final message with:
97	**SUCCESS: verified by running <describe the exact command/scenario you ran and the observed result>.**\
98	"""
99	
100	_NUDGE_2 = """\
101	STOP — you tried to finish again but you only inspected files (ls/cat/grep). That is not a functional test and will not catch a wrong value or broken behavior.
102	
103	You must EXECUTE the solution against the task's real scenario at least once before finishing:
104	- If it is a script/program: run it with the real inputs and read the actual output.
105	- If it is a fix: reproduce the original failing case and confirm it now passes.
106	- If it is a service/daemon/monitor: launch the described workload and confirm the required property holds while it runs.
107	- Recompute any required numeric constant or threshold independently and confirm the exact match.
108	
109	Do the real run now. If a genuine run confirms correctness, finish with:
110	**SUCCESS: verified by running <exact command/scenario> — observed <result>.**\
111	"""
112	
113	
114	class FunctionalVerifyGateProcessor(MultiHookProcessor):
115	    """Stateful, functional-aware exit gate.
116	
117	    Replaces the single-shot self-verify. Nudges the agent to run a real
118	    end-to-end test before exiting, and re-nudges (bounded) if the agent
119	    tries to exit again having only done read-only inspection.
120	    """
121	
122	    _singleton_group = "tb2_self_verify"  # same slot as the old gate
123	    _order = 90
124	
125	    def __init__(self, max_nudges: int = 2) -> None:
126	        self.max_nudges = max(1, int(max_nudges))
127	        self._nudges_used = 0
128	        self._substantive_since_nudge = False
129	        self._pending_message = ""
130	
131	    async def on_task_start(self, event: TaskStartEvent):
132	        self._nudges_used = 0
133	        self._substantive_since_nudge = False
134	        self._pending_message = ""
135	        yield event
136	
137	    async def on_before_model(self, event: BeforeModelEvent):
138	        if not self._pending_message:
139	            yield event
140	            return
141	        msg = self._pending_message
142	        self._pending_message = ""
143	        yield dataclasses.replace(
144	            event,
145	            messages=event.messages + (Message(role="user", content=msg),),
146	        )
147	
148	    async def on_before_tool(self, event: ToolCallEvent):
149	        # Swallow our own keepalive tool; also observe real Bash usage.
150	        if event.tool_name == _TOOL:
151	            yield dataclasses.replace(event, approved=False, synthetic_result=_ACK)
152	            return
153	        if event.tool_name == "Bash":
154	            command = ""
155	            try:
156	                command = event.tool_input.get("command", "") or ""
157	            except AttributeError:
158	                command = ""
159	            if _is_substantive(command):
160	                self._substantive_since_nudge = True
161	        yield event
162	
163	    async def on_after_model(self, event: ModelResponseEvent):
164	        exit_intent = event.finish_reason in ("end_turn", "stop") and not event.tool_calls
165	        if not exit_intent:
166	            yield event
167	            return
168	
169	        # First exit attempt: always nudge once.
170	        if self._nudges_used == 0:
171	            self._nudges_used = 1
172	            self._substantive_since_nudge = False
173	            self._pending_message = _NUDGE_1
174	            yield self._keepalive(event)
175	            return
176	
177	        # Subsequent exit attempts: allow exit if the agent has actually
178	        # exercised the solution since the last nudge, or if we've hit the cap.
179	        if self._substantive_since_nudge or self._nudges_used >= self.max_nudges:
180	            yield event
181	            return
182	
183	        # Agent tried to exit again but only did read-only inspection.
184	        self._nudges_used += 1
185	        self._substantive_since_nudge = False
186	        self._pending_message = _NUDGE_2
187	        yield self._keepalive(event)
188	
189	    def _keepalive(self, event: ModelResponseEvent):
190	        keepalive = ToolCall(
191	            id=f"fv-{uuid.uuid4().hex[:8]}",
192	            name=_TOOL,
193	            input={},
194	        )
195	        return dataclasses.replace(event, tool_calls=(keepalive,))
196	
197	    async def on_task_end(self, event: TaskEndEvent):
198	        self._nudges_used = 0
199	        self._substantive_since_nudge = False
200	        self._pending_message = ""
201	        yield event
