71	        self,
72	        command: str,
73	        cwd: str | None = None,
74	        timeout: float = 30.0,
75	    ) -> str:
76	        # Run the command inside a new session via setsid so it gets its own
77	        # process group (PGID == PID of the setsid'd process).  The PID is
78	        # saved to _PGID_FILE so kill_running() can target exactly this group
79	        # without affecting other container processes.
80	        #
81	        # stdout/stderr flow through the background job's inherited file
82	        # descriptors to docker compose exec's client as normal.  The exit
83	        # code is forwarded via `wait`.
84	        wrapped = (
85	            f"setsid bash -c {shlex.quote(command)} &"
86	            f" _hx_pid=$!;"
87	            f" printf '%s' \"$_hx_pid\" > {_PGID_FILE};"
88	            f" wait $_hx_pid"
89	        )
90	        try:
91	            result = await self._env.exec(
92	                wrapped,
93	                cwd=cwd or self._workspace_path,
94	                timeout_sec=int(timeout),
95	            )
96	        except BaseException:
97	            # Harbor raises RuntimeError on timeout (not asyncio.TimeoutError).
98	            # CancelledError arrives when an outer asyncio.wait_for fires.
99	            # Either way the setsid'd process is still running — kill it.
100	            try:
101	                await asyncio.shield(self.kill_running())
102	            except Exception:
103	                pass
104	            raise
105	        stdout = result.stdout or ""
106	        stderr = result.stderr or ""
107	        output = stdout
108	        if stderr:
109	            output = f"{output}\nSTDERR: {stderr}" if output else f"STDERR: {stderr}"
110	        if not output:
111	            output = f"(exit {result.return_code}, no output captured)"
112	        elif result.return_code != 0:
113	            output = f"{output}\n(exit {result.return_code})"
114	        return self._trim(output)
