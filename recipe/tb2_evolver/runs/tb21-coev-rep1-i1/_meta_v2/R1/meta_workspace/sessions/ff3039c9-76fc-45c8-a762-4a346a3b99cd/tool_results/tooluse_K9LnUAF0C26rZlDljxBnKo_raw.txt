672	    async def process(self, event: Event) -> AsyncIterator[Event]:
673	        method_name = self._DISPATCH.get(type(event))
674	        if not method_name:
675	            yield event
676	            return
677	        handler = getattr(self, method_name, None)
678	        if not handler:
679	            yield event
680	            return
681	        try:
682	            async for out in handler(event):
683	                yield out
684	        except asyncio.CancelledError:
685	            raise
686	        except Exception as exc:
687	            # Control-flow exceptions (budget/loop/context/parse signals) must
688	            # propagate so run_loop can translate them into exit_reason.  Any
689	            # other exception is a bug inside the processor — we log identity
690	            # (processor class + hook) and pass the event through unchanged so
691	            # one buggy processor cannot crash the run.  This is the backstop
692	            # for generated meta-skill processors whose runtime bugs escape
693	            # smoke_load + admission probe.
694	            from .runloop import HarnessError
695	
696	            if isinstance(exc, (HarnessError, ContractViolationError)):
697	                raise
698	            exc_name = type(exc).__name__
699	            if exc_name in (
700	                "ToolFailureLimitError",
701	                "ContextLengthExceeded",
702	                "GenerationAborted",
703	            ):
704	                raise
705	            _logger.warning(
706	                "processor crashed, skipping: processor=%s hook=%s event=%s error=%s: %s",
707	                type(self).__name__,
708	                method_name,
709	                type(event).__name__,
710	                exc_name,
711	                exc,
712	            )
713	            _logger.debug("processor crash traceback:", exc_info=True)
714	            yield event
