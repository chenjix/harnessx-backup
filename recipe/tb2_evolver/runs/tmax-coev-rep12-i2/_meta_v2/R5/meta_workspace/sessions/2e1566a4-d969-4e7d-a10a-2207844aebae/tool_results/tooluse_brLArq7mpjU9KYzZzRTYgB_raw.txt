3	"""CommandRepetitionRecoveryProcessor for Tmax / TB2-style agents.
4	
5	Closes a systemic loop failure that both existing guards miss:
6	
7	* ``LengthTruncationRecoveryProcessor`` fires only on
8	  ``finish_reason == "length"`` with no tool call — these loops finish
9	  normally and carry tool calls.
10	* ``ContentRepetitionRecoveryProcessor`` (R2) fingerprints assistant
11	  *narration* content; the observed R2 loops emit long, re-worded
12	  narration turn after turn (the model rephrases "I'm stuck, let me try
13	  a different approach" each time) so no two consecutive turns share the
14	  same content prefix — that guard never armed on the loop cluster.
15	* ``CustomEditToolProcessor`` counts only *file-write* commands
16	  (``cat > file`` / editor writes); these loops re-issue *diagnostic /
17	  compile / probe* commands (``curl ...``, ``g++ ...``, ``openssl ...``,
18	  ``echo ... | ...``) which it ignores.
19	
20	The robust invariant across the whole loop cluster is at the **tool-call
21	layer**: the agent re-issues the *same Bash command string* many times
22	(observed 10x–34x identical commands within a single 80-step task) while
23	producing no new state and never writing the required output. This
24	processor keys on the normalized Bash command string — orthogonal to the
25	R2 content fingerprint — and after a command has been issued
26	``repeat_threshold`` times arms an escalating corrective nudge that tells
27	the model to stop re-running that command and either gather genuinely new
28	evidence or write the required output to its exact task path.
29	
30	Detection is over a rolling recent window (``window`` most-recent
31	commands), so it also catches the *oscillating* variant where the agent
32	alternates between a small handful of commands rather than repeating one
33	verbatim back-to-back.
34	
35	Message-contract safe: mutates ``event.messages`` by at most +1 user turn
36	per armed nudge (replacing a trailing user message the run loop already
37	appended, rather than adding a second). No task-specific literals.
38	"""
39	
40	from __future__ import annotations
41	
42	import dataclasses
43	import re
44	from collections import deque
45	
46	from harnessx.core.events import (
47	    BeforeModelEvent,
48	    Message,
49	    ModelResponseEvent,
50	    TaskEndEvent,
51	    TaskStartEvent,
52	)
53	from harnessx.core.processor import MultiHookProcessor
54	
55	_WS = re.compile(r"\s+")
56	
57	_NUDGE_FIRST = (
58	    "You have now run essentially the SAME command several times and it keeps "
59	    "returning the same result — you are looping, not progressing. Do not run "
60	    "that command again. Change tactic concretely to get NEW information: inspect "
61	    "a different file or path, print intermediate values, dump raw bytes, reduce "
62	    "the problem to a smaller isolated test, or read the exact error more "
63	    "carefully. Run one command you have not run before."
64	)
65	
66	_NUDGE_REPEAT = (
67	    "STOP — you are stuck re-running the same command and burning your step "
68	    "budget with no new progress. Re-running it will not change the outcome. "
69	    "Decide right now between exactly two moves and take one: (a) if you have a "
70	    "partial but plausible result, WRITE the required output file(s) to the "
71	    "EXACT path named in the task, then confirm each exists with `ls -l`; or "
72	    "(b) gather genuinely new evidence with a single command you have not tried "
73	    "before. Do not repeat the command that is looping."
74	)
75	
76	
77	def _normalize_cmd(cmd: str) -> str:
78	    """Whitespace-normalized, length-bounded command signature.
79	
80	    Bounded so a huge here-doc file-write is compared by its leading bytes
81	    (which are stable across re-issues) rather than allocating on every call.
82	    """
83	    if not cmd:
84	        return ""
85	    return _WS.sub(" ", cmd.strip())[:400]
86	
87	
88	class CommandRepetitionRecoveryProcessor(MultiHookProcessor):
89	    """Break a repeated-Bash-command loop and redirect the agent to act."""
90	
91	    _singleton_group = "tmax_command_repetition_recovery"
92	    # After the content-repetition guard (_order=6) so the two orthogonal
93	    # detectors don't both fire on the same turn; this one keys on tool args.
94	    _order = 7
95	
96	    def __init__(
97	        self,
98	        repeat_threshold: int = 4,
99	        escalate_threshold: int = 7,
100	        window: int = 12,
101	        min_cmd_chars: int = 3,
102	    ) -> None:
103	        # Times the same command must appear in the recent window before the
104	        # first nudge. 4 is well clear of legitimate 2-3x retries (build →
105	        # fix → rebuild) seen on passing tasks.
106	        self.repeat_threshold = max(3, int(repeat_threshold))
107	        # Occurrences before the hard "stop / write output" nudge.
108	        self.escalate_threshold = max(self.repeat_threshold + 1, int(escalate_threshold))
109	        # Rolling window of most-recent commands the counter looks back over,
110	        # so oscillation between a few commands also trips.
111	        self.window = max(self.escalate_threshold, int(window))
112	        # Ignore trivially short commands (e.g. `ls`, `pwd`) so ordinary
113	        # navigation does not trip the guard.
114	        self.min_cmd_chars = max(0, int(min_cmd_chars))
115	        self._recent: deque[str] = deque(maxlen=self.window)
116	        self._pending_nudge: str = ""
117	
118	    async def on_task_start(self, event: TaskStartEvent):
119	        self._recent = deque(maxlen=self.window)
120	        self._pending_nudge = ""
121	        yield event
122	
123	    def _extract_commands(self, event: ModelResponseEvent) -> list[str]:
124	        cmds: list[str] = []
125	        for tc in event.tool_calls or ():
126	            # Only Bash tool calls carry a shell command to loop on.
127	            if getattr(tc, "name", "") not in ("Bash", "bash"):
128	                continue
129	            inp = getattr(tc, "input", None) or {}
130	            if not isinstance(inp, dict):
131	                continue
132	            raw = inp.get("command") or inp.get("cmd") or ""
133	            if not isinstance(raw, str):
134	                continue
135	            sig = _normalize_cmd(raw)
136	            if len(sig) >= self.min_cmd_chars:
137	                cmds.append(sig)
138	        return cmds
139	
140	    async def on_after_model(self, event: ModelResponseEvent):
141	        for sig in self._extract_commands(event):
142	            self._recent.append(sig)
143	            count = sum(1 for c in self._recent if c == sig)
144	            if count >= self.repeat_threshold:
145	                self._pending_nudge = (
146	                    _NUDGE_REPEAT
147	                    if count >= self.escalate_threshold
148	                    else _NUDGE_FIRST
149	                )
150	                # Drop prior occurrences of this command so we nudge once per
151	                # detected streak and re-arm only if the loop truly continues.
152	                self._recent = deque(
153	                    (c for c in self._recent if c != sig), maxlen=self.window
154	                )
155	        yield event
156	
157	    async def on_before_model(self, event: BeforeModelEvent):
158	        if not self._pending_nudge:
159	            yield event
160	            return
161	        nudge = self._pending_nudge
162	        self._pending_nudge = ""
163	        msgs = list(event.messages)
164	        # Preserve the +1 message contract: if the run loop already appended a
165	        # trailing user message, replace it rather than adding a second one.
166	        if msgs and getattr(msgs[-1], "role", None) == "user":
167	            msgs[-1] = Message(role="user", content=nudge)
168	            yield dataclasses.replace(event, messages=tuple(msgs))
169	        else:
170	            yield dataclasses.replace(
171	                event,
172	                messages=tuple(msgs) + (Message(role="user", content=nudge),),
173	            )
174	
175	    async def on_task_end(self, event: TaskEndEvent):
176	        self._recent = deque(maxlen=self.window)
177	        self._pending_nudge = ""
178	        yield event
