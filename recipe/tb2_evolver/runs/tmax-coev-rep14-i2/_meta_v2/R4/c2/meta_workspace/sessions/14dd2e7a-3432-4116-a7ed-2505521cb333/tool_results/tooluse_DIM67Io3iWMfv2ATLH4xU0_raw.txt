3	"""OcrQualityAdvisor — a TB2 control-lever MultiHookProcessor.
4	
5	Problem class this closes
6	-------------------------
7	Some TB2 tasks require reading text out of an image with ``tesseract`` (an
8	image-embedded schema, a hidden key, a config sketch, …).  The images shipped
9	with these tasks frequently carry **no DPI metadata and are physically small**,
10	so tesseract falls back to a low default resolution and emits garbled output.
11	Its own stderr flags this exactly:
12	
13	    Warning: Invalid resolution 0 dpi. Using 70 instead.
14	    Estimating resolution as 111
15	
16	An agent that trusts that first garbled read then builds a downstream artifact
17	(a parser, a detector, a mapping) on top of misread characters — the artifact
18	looks plausible but is wrong, and the verifier's exact-match / accuracy test
19	fails.  Observed twice this round: an OCR'd routing schema that produced a
20	66.6%-accurate URL mapper, and an OCR'd SSH key whose character confusions
21	(``I``/``l``/``1``, ``0``/``O``, ``5``/``S``, spurious spaces) made an
22	exact-string detector miss every adversarial sample.
23	
24	The single most effective, entirely generic tesseract remedy for this input
25	shape — **upscale the image ~3-4x and pass an explicit ``--dpi 300``, plus
26	grayscale + binarize** — was never tried in either trajectory; the agent kept
27	re-running near-identical contrast/PSM tweaks on the original resolution.
28	
29	What this processor does
30	------------------------
31	Purely mechanical, contract-safe, and general (no task-specific literals):
32	
33	* ``on_before_tool``  — remember which Bash calls actually invoked tesseract /
34	  pytesseract, so the advisory only ever attaches to a real OCR call.
35	* ``on_after_tool``   — if that call's result carries tesseract's own
36	  low-resolution signal (``Invalid resolution`` / ``Estimating resolution``),
37	  append a **one-time** advisory describing the generic high-yield
38	  preprocessing recipe and reminding the agent to cross-check ambiguous
39	  glyphs before building anything on the extracted text.
40	
41	Fires **at most once per task** and **only** when a tesseract call emitted the
42	low-resolution signal, so it is invisible to the (large) majority of non-OCR
43	tasks — zero added cost or regression surface there.  It mutates only the
44	tool-result string (the same shape as ``CustomEditToolProcessor``); it never
45	inserts, drops, or reorders messages.
46	"""
47	from __future__ import annotations
48	
49	import dataclasses
50	import re
51	
52	from harnessx.core.processor import MultiHookProcessor
53	
54	# --- detection --------------------------------------------------------------
55	
56	# The agent asked tesseract to read an image (CLI or the pytesseract binding).
57	_OCR_INVOKE_RE = re.compile(r"\btesseract\b|pytesseract|image_to_string", re.IGNORECASE)
58	
59	# Tesseract's own signal that the image lacks DPI metadata / is being read at a
60	# guessed (usually too-low) resolution — the exact case where upscale + --dpi
61	# pays off most.
62	_LOW_RES_SIGNAL_RE = re.compile(
63	    r"Invalid resolution|Estimating resolution|Using \d+ instead", re.IGNORECASE
64	)
65	
66	_ADVISORY = (
67	    "\n\n[OcrQualityAdvisor] This tesseract call reported a missing / guessed "
68	    "image resolution, which is the most common cause of garbled OCR on small "
69	    "images. Before trusting this text (or building any parser/detector/mapping "
70	    "on top of it), re-run OCR with the standard high-accuracy preprocessing you "
71	    "have not yet applied:\n"
72	    "  1. UPSCALE the image ~3-4x (e.g. PIL `img.resize((w*3, h*3), "
73	    "Image.LANCZOS)`) — this alone usually fixes small/low-DPI images.\n"
74	    "  2. Pass an EXPLICIT dpi to tesseract: `--dpi 300` (or `tesseract ... "
75	    "-c user_defined_dpi=300`).\n"
76	    "  3. Convert to grayscale and binarize (threshold) so glyph edges are "
77	    "crisp; try `--psm 6` for a uniform block of text.\n"
78	    "  4. Re-run and DIFF the new text against this one; treat characters that "
79	    "change between runs as low-confidence and disambiguate visually "
80	    "(common confusions: I/l/1, 0/O, 5/S, 8/B, rn/m, and dropped/added spaces).\n"
81	    "Verify the extracted text against any sample input/expected output the task "
82	    "provides before committing downstream code."
83	)
84	
85	
86	class OcrQualityAdvisor(MultiHookProcessor):
87	    """One-time nudge toward high-accuracy tesseract preprocessing on low-DPI OCR."""
88	
89	    _singleton_group = "ocr_quality_advisor"
90	    _order = 32  # after CustomEditToolProcessor(30), before CustomSelfVerifyProcessor(90)
91	
92	    def __init__(self) -> None:
93	        self._ocr_calls: set[str] = set()
94	        self._fired: bool = False
95	
96	    async def on_task_start(self, event):
97	        self._ocr_calls.clear()
98	        self._fired = False
99	        yield event
100	
101	    async def on_before_tool(self, event):
102	        if event.tool_name == "Bash":
103	            command = (event.tool_input or {}).get("command", "") or ""
104	            if _OCR_INVOKE_RE.search(command):
105	                self._ocr_calls.add(event.tool_call_id)
106	        yield event
107	
108	    async def on_after_tool(self, event):
109	        if self._fired:
110	            self._ocr_calls.discard(event.tool_call_id)
111	            yield event
112	            return
113	
114	        was_ocr = event.tool_call_id in self._ocr_calls
115	        self._ocr_calls.discard(event.tool_call_id)
116	
117	        result_text = event.result or ""
118	        if was_ocr and _LOW_RES_SIGNAL_RE.search(result_text):
119	            self._fired = True
120	            yield dataclasses.replace(event, result=result_text + _ADVISORY)
121	        else:
122	            yield event
123	
124	    async def on_task_end(self, event):
125	        self._ocr_calls.clear()
126	        self._fired = False
127	        yield event
