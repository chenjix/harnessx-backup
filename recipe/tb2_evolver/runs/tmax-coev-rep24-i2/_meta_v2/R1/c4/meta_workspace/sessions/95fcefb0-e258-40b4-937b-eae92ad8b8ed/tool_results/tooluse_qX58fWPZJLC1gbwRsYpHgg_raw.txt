60	def _extract_written_files(cmd: str) -> list[str]:
61	    """Return deduplicated file paths that a bash command writes to."""
62	    if ">" not in cmd and "sed" not in cmd and "tee" not in cmd:
63	        return []
64	    paths: set[str] = set()
65	    for pattern in (_REDIRECT_WRITE_RE, _SED_INPLACE_RE, _TEE_WRITE_RE):
66	        for m in pattern.finditer(cmd):
67	            p = m.group(1).strip("'\"")
68	            if p and not _SKIP_PATH_RE.match(p):
69	                paths.add(p)
70	    return list(paths)
