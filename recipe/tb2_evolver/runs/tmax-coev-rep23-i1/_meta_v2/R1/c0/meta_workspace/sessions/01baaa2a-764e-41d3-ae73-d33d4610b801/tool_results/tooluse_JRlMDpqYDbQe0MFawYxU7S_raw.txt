159	class CustomEditToolProcessor(MultiHookProcessor):
160	    """Detect repeated Bash-based file edits and nudge the model to try a different approach.
161	
162	    TB2 only exposes Bash, so write operations are detected via command patterns
163	    (redirects, sed -i, tee).  When the same file is edited more than
164	    ``threshold`` times, a warning is appended to the tool result and the
165	    per-file counter is reset.
166	    """
167	
168	    _singleton_group = "bash_edit_detector"
169	    _order = 30
170	
171	    def __init__(self, threshold: int = 7) -> None:
172	        self.threshold = threshold
173	        self._edit_counts: dict[str, int] = {}
174	        self._pending: dict[str, list[str]] = {}
175	
176	    async def on_task_start(self, event: TaskStartEvent):
177	        self._edit_counts.clear()
178	        self._pending.clear()
179	        yield event
180	
181	    async def on_before_tool(self, event: ToolCallEvent):
182	        if event.tool_name == "Bash":
183	            paths = _extract_written_files(event.tool_input.get("command", ""))
184	            if paths:
185	                self._pending[event.tool_call_id] = paths
186	        yield event
187	
188	    async def on_after_tool(self, event: ToolResultEvent):
189	        paths = self._pending.pop(event.tool_call_id, [])
190	        injections: list[str] = []
191	        for path in paths:
192	            count = self._edit_counts.get(path, 0) + 1
193	            self._edit_counts[path] = count
194	            if count > self.threshold:
195	                injections.append(_EDIT_LIMIT_WARN.format(path=path, threshold=self.threshold))
196	                self._edit_counts[path] = 0
197	        if injections:
198	            yield dataclasses.replace(event, result=(event.result or "") + "".join(injections))
199	        else:
200	            yield event
201	
202	    async def on_task_end(self, event: TaskEndEvent):
203	        self._edit_counts.clear()
204	        self._pending.clear()
205	        yield event
