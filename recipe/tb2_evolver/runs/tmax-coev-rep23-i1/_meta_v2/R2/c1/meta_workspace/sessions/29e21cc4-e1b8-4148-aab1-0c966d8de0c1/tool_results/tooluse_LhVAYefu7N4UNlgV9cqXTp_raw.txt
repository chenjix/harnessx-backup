#!/usr/bin/env python3
"""
Migration utility to parse legacy URLs and migrate to new JSON schema.
Usage: python migrate.py <input_urls.txt> <output.jsonl>
"""

import sys
import re
import json
from urllib.parse import urlparse, parse_qs
from typing import Dict, List, Optional

# Route definitions extracted from the schema image
# Format: (route_pattern, path_params, query_params)
ROUTES = [
    # Route 1: catalog/item with item_id, department, sort_order
    {
        'pattern': r'^/catalog/item/(?P<item_id>[^/]+)',
        'path_params': ['item_id'],
        'query_params': ['dept', 'sort']
    },
    # Route 2: auth/session with session_id, referrer
    {
        'pattern': r'^/auth/session/(?P<session_id>[^/]+)',
        'path_params': ['session_id'],
        'query_params': ['referrer']
    },
    # Route 3: user/prefs with user_id, theme, alerts, notifications
    {
        'pattern': r'^/user/prefs/(?P<user_id>[^/]+)',
        'path_params': ['user_id'],
        'query_params': ['theme', 'alerts', 'notifications']
    },
]

def parse_url(url: str) -> Optional[Dict]:
    """
    Parse a URL and extract parameters according to the route schema.
    Returns None if the URL doesn't match any route.
    """
    try:
        parsed = urlparse(url)
        path = parsed.path
        query_string = parsed.query
        
        # Try to match against each route
        for route in ROUTES:
            match = re.match(route['pattern'], path)
            if match:
                # Extract path parameters
                path_params = {}
                for param in route['path_params']:
                    value = match.group(param)
                    path_params[param] = value
                
                # Extract query parameters
                query_params = {}
                if query_string:
                    parsed_query = parse_qs(query_string)
                    for key, values in parsed_query.items():
                        # Take the first value if multiple, or join them
                        query_params[key] = values[0] if values else ''
                
                return {
                    'path': path_params,
                    'query': query_params
                }
    
    except Exception:
        return None

def main():
    if len(sys.argv) != 3:
        print(f"Usage: {sys.argv[0]} <input_urls.txt> <output.jsonl>")
        sys.exit(1)
    
    input_file = sys.argv[1]
    output_file = sys.argv[2]
    
    with open(input_file, 'r') as f:
        urls = [line.strip() for line in f if line.strip()]
    
    results = []
    for url in urls:
        result = parse_url(url)
        if result:
            results.append(result)
    
    with open(output_file, 'w') as f:
        for result in results:
            f.write(json.dumps(result) + '\n')

if __name__ == '__main__':
    main()
