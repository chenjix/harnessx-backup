1	"""VerifierDepPrimerProcessor — ensure the external verifier's standard
2	Python HTTP-client dependency is present in the container.
3	
4	## Why this is a harness deficiency, not a capability gap
5	
6	In this benchmark the *task agent* and the *verifier* run in separate
7	phases against the same container. The verifier's test module is injected
8	**after** the agent exits and frequently does ``import requests`` to probe
9	any HTTP/network service the task asked the agent to build. If ``requests``
10	is not importable, the test module fails to even *collect* — pytest returns
11	rc=2 (``ModuleNotFoundError: No module named 'requests'``) and the task
12	scores 0 **regardless of whether the agent's service is correct**.
13	
14	Agents in these HTTP-service tasks almost never install ``requests``
15	themselves because they implement the *server* (in C/C++/Go/Node or with
16	Python stdlib ``http.server``) and probe it with ``curl`` — they have no
17	reason to pull in the Python client. So a fully-correct solution silently
18	scores 0 purely because of a missing test-side dependency. That is an
19	environment gap the harness can close, not a reasoning error by the model.
20	
21	## Mechanism
22	
23	Deterministically inject a single, quiet, idempotent ``pip install`` of the
24	standard Python HTTP-client stack (``requests`` + its transitive deps) on
25	the first model turn of every task. The command:
26	
27	  * is a no-op when the packages are already present (``pip`` reports
28	    "already satisfied" in <1s),
29	  * writes nothing to any task output path,
30	  * suppresses its own output and can never fail the turn
31	    (``|| true``), so it cannot corrupt an otherwise-correct solution,
32	  * persists in the container for the verifier phase.
33	
34	This is general environment hardening for the whole class of
35	service/network tasks whose verifier probes the agent's endpoint from
36	Python; it is not keyed to any task id, path, or literal drawn from the
37	training trajectories.
38	
39	The processor fires **exactly once per task** and appends its install call
40	to whatever the model emits on its first turn, so it never suppresses the
41	model's own first action.
42	"""
43	
44	from __future__ import annotations
45	
46	import dataclasses
47	import uuid
48	
49	from harnessx.core.events import (
50	    ModelResponseEvent,
51	    TaskEndEvent,
52	    TaskStartEvent,
53	    ToolCall,
54	)
55	from harnessx.core.processor import MultiHookProcessor
56	
57	# Standard Python HTTP client used by external verifiers to probe services.
58	# Kept intentionally small and generic — only the client library a network
59	# test harness needs; nothing task-specific.
60	_PRIMER_PACKAGES = "requests"
61	
62	# Quiet + idempotent + non-fatal. Runs against whatever python3/pip the
63	# verifier will use. Never touches a task output path.
64	_PRIMER_CMD = (
65	    "python3 -m pip install --quiet --disable-pip-version-check "
66	    f"{_PRIMER_PACKAGES} >/dev/null 2>&1 || true"
67	)
68	
69	
70	class VerifierDepPrimerProcessor(MultiHookProcessor):
71	    """Inject a one-shot, idempotent install of the standard HTTP client so
72	    the post-agent verifier's ``import requests`` collection step succeeds.
73	
74	    Fires at most once per task, on the first model response.
75	    """
76	
77	    _singleton_group = "verifier_dep_primer"
78	    # After tool-call correction / self-verify ordering; a large order keeps
79	    # it out of the way of the message-shaping processors.
80	    _order = 32
81	
82	    def __init__(self, packages: str | None = None) -> None:
83	        # Allow override, but default to the generic HTTP client stack.
84	        self.packages = packages or _PRIMER_PACKAGES
85	        self._cmd = (
86	            "python3 -m pip install --quiet --disable-pip-version-check "
87	            f"{self.packages} >/dev/null 2>&1 || true"
88	        )
89	        self._primed = False
90	
91	    async def on_task_start(self, event: TaskStartEvent):
92	        self._primed = False
93	        yield event
94	
95	    async def on_after_model(self, event: ModelResponseEvent):
96	        if self._primed:
97	            yield event
98	            return
99	        self._primed = True
100	        primer = ToolCall(
101	            id=f"vdp-{uuid.uuid4().hex[:8]}",
102	            name="Bash",
103	            input={"command": self._cmd},
104	        )
105	        # Append the install to whatever the model already asked for so we
106	        # never suppress its own first action. If the model emitted no calls
107	        # (rare), the install still runs.
108	        yield dataclasses.replace(event, tool_calls=event.tool_calls + (primer,))
109	
110	    async def on_task_end(self, event: TaskEndEvent):
111	        self._primed = False
112	        yield event
