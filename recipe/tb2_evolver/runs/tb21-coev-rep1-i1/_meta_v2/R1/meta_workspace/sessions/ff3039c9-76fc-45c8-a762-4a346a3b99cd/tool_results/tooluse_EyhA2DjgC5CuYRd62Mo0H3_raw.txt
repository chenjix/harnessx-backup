255

258	    2. ``len(raw_track) == len(effective_track)`` (when ``raw_msgs`` provided).

260	    4. Same-index ``tool_call_id`` for ``role=tool`` messages.
261	    5. ``before_model`` net length change only 0 or +1.
262	    6. Non-authorized hooks must have net length change == 0.
263	    7. boundary consistency is handled by RunLoop (hash-based); not rechecked here.

266	    sys_count = sum(1 for m in final_msgs if m.role == "system") if final_msgs else 0

287	                hook=hook,
288	                violation_type="invariant_track_length",
289	                step_id=step_id,

296	                        hook=hook,

298	                        step_id=step_id,

304	                        hook=hook,
305	                        violation_type="invariant_tool_id_mismatch",
306	                        step_id=step_id,

315	    if hook == "before_model" and net not in (0, 1):

319	            violation_type=_VT_LENGTH_EXCEEDED,

321	        )

324	    _NO_LEN_CHANGE = {"after_model", "before_tool", "after_tool", "step_end", "task_end", "task_start"}
