3	"""FuzzyLoopTerminatorProcessor — force a clean stop on NEAR-identical turn loops.
4	
5	Motivation (R5, grounded in the R4 trajectories)
6	-------------------------------------------------
7	Every Tmax evolve round R0-R4 is flat at pass_rate 0.20. Across those rounds the
8	single dominant *harness-level* waste is a degenerate assistant-turn loop: on a
9	hard subgoal the model regenerates essentially the same "I've been stuck in a
10	loop, let me try a fundamentally different approach ..." turn over and over,
11	burning 30-60 steps and 600-1600s of wall-clock, and in the worst case the
12	accumulating truncated/length-recovery state crashes the whole run
13	(``exit_reason=error`` / ``agent_error``).
14	
15	The R1 ``BashLoopBreakerProcessor`` fingerprints only the Bash tool INPUT, so it
16	is blind to no-tool-call / narrative loops. The R4 ``DegenerateTurnTerminator``
17	fixed part of that by fingerprinting the whole assistant turn — but it requires
18	the turns to be **byte-identical**, and measurement on the R4 trajectories shows
19	the real loops DRIFT: each regenerated turn differs by a few words (a different
20	truncation point under the 4096-token cap, a reordered clause), so the
21	byte-identical consecutive run tops out at 4 and never reaches R4's
22	``terminate_threshold=6``. The R4 terminator's banner fired **zero** times on the
23	two ``exit_reason=error`` crash tasks and on the budget-exceeded loopers.
24	
25	Measured on the R4 run (max consecutive run of assistant turns whose normalized
26	token-set similarity >= 0.85):
27	
28	    FAILING loopers : 4, 5, 6, 10, 17  (task_000032, 001044, 001465, 001201, 001028)
29	    PASSING set      : max = 2 (nine of ten passers = 1)
30	
31	So a *similarity-based* run counter separates the loopers from the passing set
32	even more sharply than the exact-match one did, with a wide safety margin: a
33	terminate threshold of 5 (warn at 3) sits far above the passing-set ceiling of 2
34	yet catches every drifting loop the R4 exact matcher missed.
35	
36	What this processor does
37	------------------------
38	Fires ``on_before_model``. It keeps a short window of recent assistant-turn
39	fingerprints (a normalized token *set* per turn) and counts the current run of
40	consecutive turns that are near-duplicates of the run's anchor (similarity
41	>= ``similarity_threshold`` via Jaccard on the token set — cheap, order- and
42	truncation-insensitive, no external deps).
43	
44	* At ``warn_threshold`` near-identical turns it appends exactly one ``user``
45	  nudge (contract-safe: the last context message at that point is the repeated
46	  assistant turn, so +1 user is valid).
47	* At ``terminate_threshold`` it sets ``skip_model=True`` + ``synthetic_output``;
48	  the run loop converts that into a ``finish_reason="stop"`` response with no
49	  tool calls — its own clean-exit condition — so the task ends
50	  ``exit_reason=done`` instead of crashing to ``error`` or burning the whole
51	  budget. Partial deliverables on disk are preserved and scored, and exit-intent
52	  processors (e.g. the verifier-dep guard) still run.
53	
54	This SUBSUMES the R4 exact-match terminator (byte-identical turns have
55	similarity 1.0) while additionally catching the drifting loops it missed, so R5
56	replaces the exact matcher with this one rather than stacking both.
57	
58	Strategy-only / benchmark-agnostic: no task ids, no command literals, no domain
59	constants. Fires purely on the structural "near-identical assistant turn N times
60	in a row" shape.
61	"""
62	
63	from __future__ import annotations
64	
65	import dataclasses
66	import re
67	
68	from harnessx.core.events import (
69	    BeforeModelEvent,
70	    Message,
71	    TaskEndEvent,
72	    TaskStartEvent,
73	    _extract_text,
74	)
75	from harnessx.core.processor import MultiHookProcessor
76	
77	_WARN_TEMPLATE = (
78	    "[LoopTerminator] Your last {count} responses have been essentially the same "
79	    "with no change in outcome. Repeating this line of attack will not help. Take "
80	    "a genuinely different action this turn — inspect a different file, change the "
81	    "command, try another algorithm or input, or, if your required deliverables "
82	    "already exist at the paths the task names, verify them with `ls -lh` and "
83	    "finish. If you keep repeating near-identical responses the session will be "
84	    "ended automatically so your completed work is preserved."
85	)
86	
87	_TERMINATE_OUTPUT = (
88	    "[LoopTerminator] The session was ended automatically after {count} "
89	    "near-identical responses with no progress. Whatever work was completed "
90	    "before the loop has been preserved on disk and will be evaluated. Ending "
91	    "now to avoid wasting the remaining budget on a dead loop."
92	)
93	
94	_WORD_RE = re.compile(r"[A-Za-z0-9_]+")
95	
96	
97	class FuzzyLoopTerminatorProcessor(MultiHookProcessor):
98	    """Detect near-identical repeated assistant turns and force a clean stop.
99	
100	    Args:
101	        similarity_threshold: Jaccard similarity on the normalized token set of
102	            two assistant turns at/above which they count as "the same" turn
103	            (default 0.85). Byte-identical turns score 1.0, so this subsumes
104	            exact matching.
105	        warn_threshold: consecutive near-identical turns at which a single user
106	            nudge is injected (default 3). Must be >= 2.
107	        terminate_threshold: consecutive near-identical turns at which the
108	            session is force-terminated via skip_model + synthetic_output
109	            (default 5). Must be > warn_threshold. Kept well above the observed
110	            passing-set ceiling (max run = 2) for a wide safety margin.
111	        min_tokens: turns with fewer than this many tokens are ignored for
112	            similarity (too short to compare meaningfully; resets the run).
113	    """
114	
115	    _singleton_group = "fuzzy_loop_terminator"
116	    # After compaction / token-budget so the fingerprint reflects the context the
117	    # model will actually see this step; before the model call.
118	    _order = 23
119	
120	    def __init__(
121	        self,
122	        similarity_threshold: float = 0.85,
123	        warn_threshold: int = 3,
124	        terminate_threshold: int = 5,
125	        min_tokens: int = 5,
126	    ) -> None:
127	        self.similarity_threshold = float(similarity_threshold)
128	        self.warn_threshold = max(2, int(warn_threshold))
129	        self.terminate_threshold = max(self.warn_threshold + 1, int(terminate_threshold))
130	        self.min_tokens = max(1, int(min_tokens))
131	        self._anchor: frozenset[str] | None = None
132	        self._run: int = 0
133	        self._terminated: bool = False
134	
135	    # ------------------------------------------------------------------
136	    def _reset(self) -> None:
137	        self._anchor = None
138	        self._run = 0
139	        self._terminated = False
140	
141	    @staticmethod
142	    def _token_set(text: str) -> frozenset[str]:
143	        return frozenset(_WORD_RE.findall((text or "").lower()))
144	
145	    @staticmethod
146	    def _jaccard(a: frozenset[str], b: frozenset[str]) -> float:
147	        if not a and not b:
148	            return 1.0
149	        if not a or not b:
150	            return 0.0
151	        inter = len(a & b)
152	        union = len(a | b)
153	        return inter / union if union else 0.0
154	
155	    @staticmethod
156	    def _last_assistant_text(messages: tuple) -> str:
157	        for m in reversed(messages):
158	            role = getattr(m, "role", None)
159	            if role == "assistant":
160	                return _extract_text(getattr(m, "content", "")) or ""
161	            # Skip trailing tool/user messages injected after the assistant turn.
162	        return ""
163	
164	    # ------------------------------------------------------------------
165	    async def on_task_start(self, event: TaskStartEvent):
166	        self._reset()
167	        yield event
168	
169	    async def on_task_end(self, event: TaskEndEvent):
170	        self._reset()
171	        yield event
172	
173	    async def on_before_model(self, event: BeforeModelEvent):
174	        if self._terminated:
175	            yield event
176	            return
177	
178	        text = self._last_assistant_text(event.messages)
179	        toks = self._token_set(text)
180	
181	        if len(toks) < self.min_tokens:
182	            # No usable assistant turn yet, or too short to compare — reset run.
183	            self._anchor = None
184	            self._run = 0
185	            yield event
186	            return
187	
188	        if self._anchor is not None and self._jaccard(self._anchor, toks) >= self.similarity_threshold:
189	            self._run += 1
190	        else:
191	            self._anchor = toks
192	            self._run = 1
193	
194	        # Hard terminate: force a clean stop. The run loop converts skip_model
195	        # into a finish_reason="stop" response with no tool calls -> clean break
196	        # -> exit_reason="done".
197	        if self._run >= self.terminate_threshold:
198	            self._terminated = True
199	            yield dataclasses.replace(
200	                event,
201	                skip_model=True,
202	                synthetic_output=_TERMINATE_OUTPUT.format(count=self._run),
203	            )
204	            return
205	
206	        # Warn: append exactly one user message (contract-safe; the last context
207	        # message here is the repeated assistant turn, so +1 user is valid).
208	        if self._run >= self.warn_threshold:
209	            warn = _WARN_TEMPLATE.format(count=self._run)
210	            yield dataclasses.replace(
211	                event,
212	                messages=event.messages + (Message(role="user", content=warn),),
213	            )
214	            return
215	
216	        yield event
