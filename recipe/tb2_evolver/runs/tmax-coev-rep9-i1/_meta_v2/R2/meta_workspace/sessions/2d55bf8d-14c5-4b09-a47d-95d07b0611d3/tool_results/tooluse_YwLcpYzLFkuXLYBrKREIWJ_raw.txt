129	        """Reset all per-task state at the beginning of each task."""
130	        self._reset()
131	        self._current_run_id = event.run_id
132	        yield event
133	
134	    async def on_step_start(self, event: StepStartEvent):
135	        """Detect compaction-driven message drops; clear stale fingerprints if needed."""
136	        if event.run_id != self._current_run_id:
137	            self._reset()
138	            self._current_run_id = event.run_id
139	
140	        # CompactionProcessor (order=8) runs before us (order=20) so
141	        # event.messages already reflects any eviction done this step.
142	        current_count = len(event.messages)
143	        if self._prev_message_count > 0 and self._prev_message_count - current_count >= self.compaction_drop_threshold:
144	            self._fingerprints.clear()
145	            self._name_fingerprints.clear()
146	            self._pending_fp.clear()
147	        self._prev_message_count = current_count
148	        yield event
149	
150	    async def on_before_tool(self, event: ToolCallEvent):
151	        """Capture the exact fingerprint (name + serialised inputs) for this call.
152	
153	        Stored by tool_call_id so on_after_tool can look it up without needing
154	        the inputs again — ToolResultEvent does not carry them.
155	        """
156	        try:
157	            input_str = json.dumps(event.tool_input, sort_keys=True, ensure_ascii=False)
158	        except Exception:
159	            input_str = repr(event.tool_input)
160	        self._pending_fp[event.tool_call_id] = self._compute_fingerprint(f"{event.tool_name}\x00{input_str}")
161	        yield event
162	
163	    async def on_after_tool(self, event: ToolResultEvent):
164	        """Detect loop and inject warning directly into this tool result if needed.
165	
166	        Both strategies are evaluated here, so no staging step is required.
167	        The warning is appended to the result content so the system prompt is
168	        never touched by this processor.
169	        """
170	        fp = self._pending_fp.pop(event.tool_call_id, None)
171	
172	        # Steps with no fingerprint (e.g. on_before_tool not seen) break the run.
173	        if not fp:
174	            self._fingerprints.append("")
175	            self._name_fingerprints.append("")
176	            yield event
177	            return
178	
179	        # ── Strategy 1: exact fingerprint (name + inputs) ──────────────
180	        s1_run = self._consecutive_tail(self._fingerprints, fp) + 1
181	        self._fingerprints.append(fp)
182	
183	        if s1_run >= self.threshold:
184	            raise LoopDetectedError(
185	                f"Loop detected: identical tool call '{event.tool_name}' repeated {s1_run} times consecutively"
186	            )
187	
188	        warning = ""
189	        if s1_run >= self.warn_threshold:
190	            warning = _WARN_TEMPLATE.format(
191	                count=s1_run,
192	                tools=f"`{event.tool_name}`",
193	            )
194	
195	        # ── Strategy 2: name-only fingerprint (warn-only, no raise) ────
196	        s2_run = self._consecutive_tail(self._name_fingerprints, event.tool_name) + 1
197	        self._name_fingerprints.append(event.tool_name)
198	
199	        if not warning and s2_run >= self.name_warn_threshold:
200	            warning = _NAME_WARN_TEMPLATE.format(
201	                tool=event.tool_name,
202	                count=s2_run,
203	                pattern=event.tool_name,
204	            )
205	
206	        if warning:
207	            yield dataclasses.replace(event, result=(event.result or "") + "\n\n" + warning)
208	        else:
209	            yield event
210	
211	    async def on_task_end(self, event: TaskEndEvent):
212	        self._reset()
213	        yield event
