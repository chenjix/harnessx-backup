3	from __future__ import annotations
4	
5	from ...core.events import ModelResponseEvent, TaskEndEvent
6	from ...core.processor import MultiHookProcessor
7	from ...core.runloop import ModelParseError
8	
9	
10	class ParseRetryProcessor(MultiHookProcessor):
11	    """Validate ModelResponseEvent structure and raise on malformed output.
12	
13	    Checks that every tool call in the model response has a non-empty name and
14	    a dict input.  Raises :exc:`ModelParseError` on the first violation.
15	
16	    Args:
17	        max_consecutive_errors: Raise only after this many consecutive bad
18	            responses (default 1 — fail on first).  Set higher to tolerate
19	            occasional model hiccups before aborting.
20	    """
21	
22	    _singleton_group = "parse_retry"
23	    _order = 10
24	
25	    def __init__(self, max_consecutive_errors: int = 1) -> None:
26	        self.max_consecutive_errors = max_consecutive_errors
27	        # Keyed by run_id so parallel workers (sharing the singleton via
28	        # _singleton_group) each track their own consecutive-error count.
29	        self._error_counts: dict[str, int] = {}
30	
31	    def _is_valid(self, event: ModelResponseEvent) -> tuple[bool, str]:
32	        """Return (valid, error_message)."""
33	        for tc in event.tool_calls:
34	            if not tc.name:
35	                return False, "tool_call missing name"
36	            if not isinstance(tc.input, dict):
37	                return False, f"tool_call '{tc.name}' input is not a dict"
38	        return True, ""
39	
40	    async def on_after_model(self, event: ModelResponseEvent):
41	        run_id = event.run_id
42	        valid, err = self._is_valid(event)
43	        if valid:
44	            self._error_counts.pop(run_id, None)
45	            yield event
46	            return
47	
48	        count = self._error_counts.get(run_id, 0) + 1
49	        self._error_counts[run_id] = count
50	        if count >= self.max_consecutive_errors:
51	            self._error_counts.pop(run_id, None)
52	            raise ModelParseError(f"Malformed model response ({count} consecutive): {err}")
53	        yield event
54	
55	    async def on_task_end(self, event: TaskEndEvent):
56	        self._error_counts.pop(event.run_id, None)
57	        yield event
