3	"""GraderClientDepProcessor — ensure common grader HTTP-client deps are present.
4	
5	Motivation (harness deficiency, not a task solution)
6	-----------------------------------------------------
7	Several Tmax tasks ask the agent to build and run a local HTTP service
8	(Flask / FastAPI / http.server / a compiled binary bound to a port). The
9	external verifier phase — which runs *after* the agent exits and whose test
10	files are not visible during the agent phase — hits that service using the
11	``requests`` library. When ``requests`` is not installed in the image, the
12	verifier's ``test_*.py`` fails at *collection* time with::
13	
14	    ModuleNotFoundError: No module named 'requests'
15	
16	i.e. the task is graded 0 even when the agent's service is fully correct.
17	Agents in this benchmark consistently probe their own service with
18	``urllib``/``curl`` (both preinstalled), so they never surface — and never
19	fix — the missing ``requests`` dependency the grader depends on.
20	
21	This is a structural gap: the agent has no signal that the grader will
22	``import requests``, and the verifier's test file is absent during the run.
23	The fix is a mechanical, idempotent environment-preparation step, not domain
24	knowledge — hence a Control processor rather than a prompt rule.
25	
26	Mechanism
27	---------
28	* ``on_after_tool`` — watch Bash commands for HTTP-service signatures
29	  (framework imports, port binds, ``127.0.0.1:<port>``, ``manage.py
30	  runserver`` …). If seen, arm the guard.
31	* ``on_after_model`` — when the model tries to end the turn (no tool calls)
32	  and the guard is armed but not yet satisfied, inject a single real Bash
33	  tool call that best-effort installs the common grader client libs. The
34	  install is quiet, time-bounded, and idempotent: if the packages are
35	  already present (or the network is unavailable) it is a harmless no-op and
36	  the agent proceeds to exit on the next turn.
37	
38	Fires at most once per task. Scoped to a *class* of tasks (any task that
39	stands up an HTTP service) — no task ids, no dataset-specific literals.
40	"""
41	from __future__ import annotations
42	
43	import dataclasses
44	import re
45	import uuid
46	
47	from harnessx.core.events import (
48	    ModelResponseEvent,
49	    TaskEndEvent,
50	    TaskStartEvent,
51	    ToolCall,
52	    ToolCallEvent,
53	)
54	from harnessx.core.processor import MultiHookProcessor
55	
56	# Signatures that indicate the agent is standing up a local HTTP service.
57	# Kept generic across web frameworks + stdlib + raw socket binds.
58	_HTTP_SERVICE_RE = re.compile(
59	    r"(?:"
60	    r"\bflask\b|\bfastapi\b|\buvicorn\b|\bgunicorn\b|\bwaitress\b|\bhypercorn\b"
61	    r"|\bhttp\.server\b|\bBaseHTTPRequestHandler\b|\bHTTPServer\b|\bsocketserver\b"
62	    r"|\bmanage\.py\s+runserver\b|\bapp\.run\("
63	    r"|\b127\.0\.0\.1:\d+\b|\b0\.0\.0\.0:\d+\b|\blocalhost:\d+\b"
64	    r"|--port\s*=?\s*\d+|\blisten\s+\d+|\bbind\(\s*\(\s*['\"]"
65	    r")",
66	    re.IGNORECASE,
67	)
68	
69	# Best-effort, quiet, idempotent install of the HTTP client libs a grader
70	# commonly imports to exercise a service. If already installed or the network
71	# is down, this is a no-op that costs a couple of seconds.
72	_INSTALL_CMD = (
73	    "(python3 -c 'import requests' 2>/dev/null && echo '[grader-deps] requests already present') "
74	    "|| (pip install --quiet --disable-pip-version-check requests >/dev/null 2>&1 "
75	    "|| pip3 install --quiet --disable-pip-version-check requests >/dev/null 2>&1); "
76	    "python3 -c 'import requests, sys; print(\"[grader-deps] requests \" + requests.__version__)' "
77	    "2>/dev/null || echo '[grader-deps] requests unavailable (offline?) — continuing'"
78	)
79	
80	_ACK_UNUSED = None  # (install feedback is surfaced via the Bash tool result itself)
81	
82	
83	class GraderClientDepProcessor(MultiHookProcessor):
84	    """Ensure ``requests`` is installed once, for tasks that stand up a service."""
85	
86	    _singleton_group = "grader_client_deps"
87	    _order = 88  # just before CustomSelfVerifyProcessor (_order=90)
88	
89	    def __init__(self, install_timeout_s: int = 90) -> None:
90	        self.install_timeout_s = int(install_timeout_s)
91	        self._service_seen = False
92	        self._done = False
93	
94	    async def on_task_start(self, event: TaskStartEvent):
95	        self._service_seen = False
96	        self._done = False
97	        yield event
98	
99	    async def on_before_tool(self, event: ToolCallEvent):
100	        # Arm when the agent runs a command that looks like an HTTP service.
101	        # Detected on the call event (which carries tool_input); ToolResultEvent
102	        # has no tool_input, so detection must happen here.
103	        if not self._service_seen and event.tool_name == "Bash":
104	            inp = event.tool_input if isinstance(event.tool_input, dict) else {}
105	            cmd = str(inp.get("command", "") or "")
106	            if _HTTP_SERVICE_RE.search(cmd):
107	                self._service_seen = True
108	        yield event
109	
110	    async def on_after_model(self, event: ModelResponseEvent):
111	        exit_intent = event.finish_reason in ("end_turn", "stop") and not event.tool_calls
112	        if exit_intent and self._service_seen and not self._done:
113	            self._done = True
114	            install_call = ToolCall(
115	                id=f"gdep-{uuid.uuid4().hex[:8]}",
116	                name="Bash",
117	                input={"command": _INSTALL_CMD, "timeout": self.install_timeout_s * 1000},
118	            )
119	            yield dataclasses.replace(event, tool_calls=(install_call,))
120	        else:
121	            yield event
122	
123	    async def on_task_end(self, event: TaskEndEvent):
124	        self._service_seen = False
125	        self._done = False
126	        yield event
