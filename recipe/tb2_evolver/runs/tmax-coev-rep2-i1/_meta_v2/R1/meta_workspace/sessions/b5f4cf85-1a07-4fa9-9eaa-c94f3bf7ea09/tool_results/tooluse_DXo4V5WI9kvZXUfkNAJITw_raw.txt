1	"""Thin OpenAI/vLLM agent loop with a single Bash tool (qwen3_xml)."""
2	from __future__ import annotations
3	
4	import json
5	import time
6	import urllib.error
7	import urllib.request
8	from dataclasses import dataclass, field
9	from typing import Any, Callable
10	
11	BASH_TOOL: dict[str, Any] = {
12	    "type": "function",
13	    "function": {
14	        "name": "Bash",
15	        "description": (
16	            "Execute a shell command inside the task container and return "
17	            "stdout+stderr. Prefer working under /home/user."
18	        ),
19	        "parameters": {
20	            "type": "object",
21	            "properties": {
22	                "command": {
23	                    "type": "string",
24	                    "description": "Shell command to run.",
25	                },
26	                "timeout": {
27	                    "type": "integer",
28	                    "description": "Timeout in milliseconds (default 30000).",
29	                },
30	            },
31	            "required": ["command"],
32	        },
33	    },
34	}
35	
36	SYSTEM_PROMPT = """\
37	You are a terminal coding agent solving a single Linux task.
38	Use the Bash tool to inspect the environment, edit files, and run commands.
39	Work under /home/user unless the instruction says otherwise.
40	When the task is complete, stop calling tools and briefly confirm what you did.
41	Do not ask questions — act.
42	"""
43	
44	
45	@dataclass
46	class StepLog:
47	    step: int
48	    role: str
49	    content: str | None = None
50	    tool_calls: list[dict[str, Any]] = field(default_factory=list)
51	    tool_results: list[dict[str, Any]] = field(default_factory=list)
52	
53	
54	@dataclass
55	class AgentResult:
56	    steps: int
57	    finished: str  # no_tool_calls | max_steps | error
58	    messages: list[dict[str, Any]]
59	    log: list[StepLog]
60	    error: str | None = None
61	
62	
63	def _chat(
64	    api_base: str,
65	    model: str,
66	    messages: list[dict[str, Any]],
67	    *,
68	    temperature: float,
69	    api_key: str,
70	    timeout: float,
71	    max_tokens: int | None = 4096,
72	) -> dict[str, Any]:
73	    url = api_base.rstrip("/") + "/chat/completions"
74	    body: dict[str, Any] = {
75	        "model": model,
76	        "messages": messages,
77	        "tools": [BASH_TOOL],
78	        "tool_choice": "auto",
79	        "temperature": temperature,
80	    }
81	    if max_tokens is not None and max_tokens > 0:
82	        body["max_tokens"] = int(max_tokens)
83	    data = json.dumps(body).encode()
84	    req = urllib.request.Request(
85	        url,
86	        data=data,
87	        headers={
88	            "Content-Type": "application/json",
89	            "Authorization": f"Bearer {api_key or 'EMPTY'}",
90	        },
91	        method="POST",
92	    )
93	    with urllib.request.urlopen(req, timeout=timeout) as resp:
94	        return json.loads(resp.read().decode())
95	
96	
97	def _parse_args(raw: Any) -> dict[str, Any]:
98	    if isinstance(raw, dict):
99	        return raw
100	    if not raw:
101	        return {}
102	    try:
103	        return json.loads(raw)
104	    except Exception:
105	        return {"command": str(raw)}
106	
107	
108	def run_agent(
109	    *,
110	    api_base: str,
111	    model: str,
112	    instruction: str,
113	    exec_fn: Callable[[str, float], tuple[int, str]],
114	    max_steps: int = 80,
115	    temperature: float = 0.0,
116	    api_key: str = "EMPTY",
117	    request_timeout: float = 180.0,
118	    default_cmd_timeout: float = 60.0,
119	    system_prompt: str | None = None,
120	    max_tokens: int | None = 4096,
121	) -> AgentResult:
122	    sys_prompt = (system_prompt or SYSTEM_PROMPT).strip() or SYSTEM_PROMPT
123	    messages: list[dict[str, Any]] = [
124	        {"role": "system", "content": sys_prompt},
125	        {"role": "user", "content": instruction},
126	    ]
127	    log: list[StepLog] = []
128	
129	    for step in range(1, max_steps + 1):
130	        try:
131	            resp = _chat(
132	                api_base,
133	                model,
134	                messages,
135	                temperature=temperature,
136	                api_key=api_key,
137	                timeout=request_timeout,
138	                max_tokens=max_tokens,
139	            )
140	        except Exception as e:
141	            return AgentResult(
142	                steps=step - 1,
143	                finished="error",
144	                messages=messages,
145	                log=log,
146	                error=f"chat failed at step {step}: {e}",
147	            )
148	
149	        choice = (resp.get("choices") or [{}])[0]
150	        msg = choice.get("message") or {}
151	        tool_calls = msg.get("tool_calls") or []
152	        content = msg.get("content")
153	
154	        # Normalize assistant message for history.
155	        asst: dict[str, Any] = {"role": "assistant", "content": content or ""}
156	        if tool_calls:
157	            asst["tool_calls"] = tool_calls
158	        messages.append(asst)
159	
160	        entry = StepLog(step=step, role="assistant", content=content, tool_calls=tool_calls)
161	        if not tool_calls:
162	            log.append(entry)
163	            return AgentResult(
164	                steps=step, finished="no_tool_calls", messages=messages, log=log
165	            )
166	
167	        for tc in tool_calls:
168	            fn = (tc.get("function") or {})
169	            name = fn.get("name") or ""
170	            args = _parse_args(fn.get("arguments"))
171	            tc_id = tc.get("id") or f"call_{step}"
172	            if name.lower() not in ("bash",):
173	                result = f"Error: unknown tool '{name}'. Only Bash is available."
174	                rc = 1
175	            else:
176	                cmd = str(args.get("command") or "")
177	                timeout_ms = args.get("timeout")
178	                try:
179	                    t_sec = (
180	                        min(float(timeout_ms) / 1000.0, 300.0)
181	                        if timeout_ms is not None
182	                        else default_cmd_timeout
183	                    )
184	                except Exception:
185	                    t_sec = default_cmd_timeout
186	                if not cmd.strip():
187	                    rc, result = 1, "Error: empty command"
188	                else:
189	                    t0 = time.time()
190	                    try:
191	                        rc, result = exec_fn(cmd, t_sec)
192	                    except Exception as e:
193	                        rc, result = 1, f"Error executing command: {e}"
194	                    # annotate duration for debugging
195	                    result = (result or "")[-12000:]
196	                    if not result.endswith("\n"):
197	                        result += "\n"
198	                    result += f"[rc={rc} elapsed={time.time()-t0:.1f}s]"
199	
200	            tool_msg = {
201	                "role": "tool",
202	                "tool_call_id": tc_id,
203	                "name": "Bash",
204	                "content": result,
205	            }
206	            messages.append(tool_msg)
207	            entry.tool_results.append(
208	                {"tool_call_id": tc_id, "command": args.get("command"), "rc": rc, "output_tail": result[-2000:]}
209	            )
210	        log.append(entry)
211	
212	    return AgentResult(
213	        steps=max_steps, finished="max_steps", messages=messages, log=log
214	    )
