bool isPathSafe(const std::string& targetDir, const std::string& entryPath) {
    std::string target = targetDir;
    if (!target.empty() && target.back() != '/') target += '/';
    std::string fullPath = target + entryPath;
    std::string normalized;
    std::istringstream iss(fullPath);
    std::string component;
    while (std::getline(iss, component, '/')) {
        if (component.empty() || component == ".") continue;
        if (component == "..") {
            if (normalized.empty()) return false;
            size_t lastSlash = normalized.rfind('/');
            if (lastSlash == std::string::npos) return false;
            normalized = normalized.substr(0, lastSlash);
        } else {
            if (!normalized.empty()) normalized += '/';
            normalized += component;
        }
    }
    if (normalized.size() < target.size()) return false;
    return normalized.substr(0, target.size()) == target;
}

bool createDirectory(const std::string& path) {
    std::string current;
    std::istringstream iss(path);
    std::string component;
    while (std::getline(iss, component, '/')) {
        if (component.empty()) continue;
        current += '/' + component;
        struct stat st;
        if (stat(current.c_str(), &st) != 0) {
            if (mkdir(current.c_str(), 0755) != 0 && errno != EEXIST) {
                std::cerr << "Error creating directory: " << current << std::endl;
                return false;
            }
        }
    }
    return true;
}
