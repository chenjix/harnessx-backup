771	    event: Event,
772	    processors: "list[Processor]",
773	    tracer: "object | None" = None,
774	    hook: str = "",
775	) -> "list[Event]":
776	    """Pass *event* through processors; return **every** yielded event.
777	
778	    Unlike ``pipe()``, no intermediate events are discarded.  Necessary when a
779	    hook point may produce heterogeneous output — for example when a custom
780	    ``after_model`` processor yields a ``ModelResponseEvent`` plus one or more
781	    ``SpawnSubAgentEvent`` values.
782	
783	    Returns ``[event]`` when *processors* is empty (pass-through).
784	    """
785	    chain = ProcessorChain(*processors)
786	    return [ev async for ev in chain.process(event, tracer=tracer, hook=hook)]
787	
788	
789	# ─── Hook Decorators (syntax sugar for single-node Processors) ─────────────────
790	
791	
792	def _make_hook_processor(fn: Callable, event_type: type) -> Processor:
793	    """Wrap an async generator function into a Processor object."""
794	
795	    class HookProcessor:
796	        async def process(self, event: Event) -> AsyncIterator[Event]:
797	            if isinstance(event, event_type):
798	                async for out in fn(event):
799	                    yield out
800	            else:
801	                yield event
802	
803	    HookProcessor.__name__ = getattr(fn, "__name__", "HookProcessor")
804	    return HookProcessor()
805
