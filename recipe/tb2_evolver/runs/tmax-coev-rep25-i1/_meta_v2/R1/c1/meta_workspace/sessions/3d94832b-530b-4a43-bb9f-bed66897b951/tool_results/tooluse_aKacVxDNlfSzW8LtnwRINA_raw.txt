29	_DEFAULT_SUMMARY_MAX_MSGS = 15
30	_DEFAULT_SUMMARY_MSG_CHARS = 150
31	
32	
33	async def _default_summarize(messages: list[Message]) -> str:
34	    """Fallback summariser used when no ``summarize`` sub-harness is registered.
35	
36	    Takes the most recent ``_DEFAULT_SUMMARY_MAX_MSGS`` of the evicted set
37	    (closest to the retention window, therefore most relevant) and truncates
38	    each to ``_DEFAULT_SUMMARY_MSG_CHARS`` chars.  This bounds summary size to
39	    ~15 × 150 chars regardless of how many messages were evicted, preventing
40	    unbounded accumulation across repeated compaction cycles.
41	    """
42	    tail = messages[-_DEFAULT_SUMMARY_MAX_MSGS:]
43	    parts = [f"[{m.role}] {_extract_text(m.content)[:_DEFAULT_SUMMARY_MSG_CHARS]}" for m in tail]
44	    return "[Summary of earlier context]\n" + "\n".join(parts)
