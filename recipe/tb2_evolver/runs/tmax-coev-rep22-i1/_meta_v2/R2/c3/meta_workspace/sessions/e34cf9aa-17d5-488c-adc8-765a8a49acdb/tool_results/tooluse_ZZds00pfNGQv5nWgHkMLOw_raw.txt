#!/usr/bin/env python3
"""
Deployment Monitor Daemon

Monitors disk usage in /home/user/logs/ and controls worker processes
to prevent disk quota exhaustion during deployment.
"""

import os
import time
import signal
import subprocess
import glob

LOGS_DIR = "/home/user/logs"
LOG_PATTERN = os.path.join(LOGS_DIR, "*.log")
THRESHOLD_BYTES = 40 * 1024 * 1024  # 40 MB
SLEEP_INTERVAL = 0.1


def get_total_logs_size():
    """Calculate total size of all .log files in the logs directory."""
    total_size = 0
    try:
        for filepath in glob.glob(LOG_PATTERN):
            try:
                total_size += os.path.getsize(filepath)
            except OSError:
                pass
    except OSError:
        pass
    return total_size


def get_worker_processes():
    """Get PIDs of all running worker_sim.py processes."""
    worker_pids = []
    try:
        result = subprocess.run(
            ["ps", "aux"],
            capture_output=True,
            text=True,
            timeout=5
        )
        for line in result.stdout.splitlines():
            if "worker_sim.py" in line:
                # PID is in the second column
                parts = line.split()
                if len(parts) >= 2:
                    try:
                        pid = int(parts[1])
                        worker_pids.append(pid)
                    except ValueError:
                        pass
    except (subprocess.TimeoutExpired, OSError):
        pass
    return worker_pids


def send_signal_to_workers(signum):
    """Send a signal to all worker processes."""
    worker_pids = get_worker_processes()
    for pid in worker_pids:
        try:
            os.kill(pid, signum)
        except OSError:
            # Process may have exited
            pass


def truncate_log_files():
    """Truncate all .log files in the logs directory to 0 bytes."""
    for filepath in glob.glob(LOG_PATTERN):
        try:
            with open(filepath, 'w') as f:
                pass  # Truncate to 0 bytes
        except OSError:
            pass


def main():
    """Main monitoring loop."""
    while True:
        total_size = get_total_logs_size()
        
        if total_size > THRESHOLD_BYTES:
            # Disk usage exceeds threshold, take action
            # 1. Send SIGSTOP to pause workers
            send_signal_to_workers(signal.SIGSTOP)
            
            # 2. Truncate all log files to free space
            truncate_log_files()
            
            # 3. Send SIGCONT to resume workers
            send_signal_to_workers(signal.SIGCONT)
        
        # Check if all workers have exited
        worker_pids = get_worker_processes()
        if not worker_pids:
            # No more workers running, exit cleanly
            break
        
        # Sleep to avoid high CPU usage
        time.sleep(SLEEP_INTERVAL)


if __name__ == "__main__":
    main()
