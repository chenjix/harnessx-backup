3	"""System-prompt builders for the Tmax HarnessX path.
4	
5	``SiblingSystemPromptBuilder`` reads ``system_prompt.txt`` next to the active
6	HarnessConfig YAML (resolved via ``TB2_HARNESS_CONFIG`` /
7	``TMAX_HARNESS_CONFIG``). MetaAgent may edit that sidecar freely; it may also
8	replace this builder with a Template / static builder in ``config.yaml``.
9	"""
10	from __future__ import annotations
11	
12	import os
13	from pathlib import Path
14	from typing import TYPE_CHECKING
15	
16	if TYPE_CHECKING:
17	    from harnessx.workspace.workspace import Workspace
18	
19	DEFAULT_TMAX_PROMPT = """\
20	You are a terminal coding agent solving a single Linux task.
21	Use the Bash tool to inspect the environment, edit files, and run commands.
22	Work under /home/user unless the instruction says otherwise.
23	When the task is complete, stop calling tools and briefly confirm what you did.
24	Do not ask questions — act.
25	"""
26	
27	
28	def _resolve_prompt_path() -> Path | None:
29	    for key in ("TB2_HARNESS_CONFIG", "TMAX_HARNESS_CONFIG"):
30	        raw = (os.environ.get(key) or "").strip()
31	        if not raw:
32	            continue
33	        sibling = Path(raw).expanduser().resolve().parent / "system_prompt.txt"
34	        if sibling.is_file():
35	            return sibling
36	    explicit = (os.environ.get("TMAX_SYSTEM_PROMPT_FILE") or "").strip()
37	    if explicit and Path(explicit).is_file():
38	        return Path(explicit).resolve()
39	    return None
40	
41	
42	class SiblingSystemPromptBuilder:
43	    """Load ``system_prompt.txt`` beside the evolved HarnessConfig."""
44	
45	    async def build(self, workspace: "Workspace | None" = None) -> str:  # noqa: ARG002
46	        path = _resolve_prompt_path()
47	        if path is not None:
48	            text = path.read_text(encoding="utf-8").strip()
49	            if text:
50	                return text
51	        return DEFAULT_TMAX_PROMPT.strip()
52	
53	
54	class StaticTmaxSystemPromptBuilder:
55	    """Fixed default prompt (fallback when no sidecar exists)."""
56	
57	    def __init__(self, text: str | None = None) -> None:
58	        self._text = (text or DEFAULT_TMAX_PROMPT).strip()
59	
60	    async def build(self, workspace: "Workspace | None" = None) -> str:  # noqa: ARG002
61	        return self._text
62	
63	
64	__all__ = [
65	    "DEFAULT_TMAX_PROMPT",
66	    "SiblingSystemPromptBuilder",
67	    "StaticTmaxSystemPromptBuilder",
68	]
