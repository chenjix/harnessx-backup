3	"""OcrQualityAdvisor — a TB2 control-lever MultiHookProcessor.
4	
5	Problem class this closes
6	-------------------------
7	Some TB2 tasks require reading text out of an image with ``tesseract`` (an
8	image-embedded schema, a hidden key, a config sketch, …).  The images shipped
9	with these tasks frequently carry **no DPI metadata and are physically small**,
10	so tesseract falls back to a low default resolution and emits garbled output.
11	Its own stderr flags this exactly:
12	
13	    Warning: Invalid resolution 0 dpi. Using 70 instead.
14	    Estimating resolution as 111
15	
16	An agent that trusts that first garbled read then builds a downstream artifact
17	(a parser, a detector, a mapping) on top of misread characters — the artifact
18	looks plausible but is wrong, and the verifier's exact-match / accuracy test
19	fails.  Observed twice this trajectory set: an OCR'd routing schema that
20	produced a 65.6%-accurate URL mapper (task_000015; verifier threshold 0.98),
21	and an OCR'd SSH key whose character confusions (``I``/``l``/``1``, ``0``/``O``,
22	``5``/``S``, spurious spaces) made an exact-string detector miss every
23	adversarial sample (task_000505; "2 of 2 evil bypassed").
24	
25	The single most effective, entirely generic tesseract remedy for this input
26	shape — **upscale the image ~3-4x and pass an explicit ``--dpi 300``, plus
27	grayscale + binarize** — was never tried in either trajectory; the agent kept
28	re-running near-identical contrast/PSM tweaks on the original resolution.
29	
30	What this processor does
31	------------------------
32	Purely mechanical, contract-safe, and general (no task-specific literals):
33	
34	* ``on_before_tool``  — remember which Bash calls actually invoked tesseract /
35	  pytesseract, so the advisory only ever attaches to a real OCR call.
36	* ``on_after_tool``   — if that call's result carries tesseract's own
37	  low-resolution signal (``Invalid resolution`` / ``Estimating resolution``),
38	  append a **one-time** advisory describing the generic high-yield
39	  preprocessing recipe and reminding the agent to cross-check ambiguous
40	  glyphs before building anything on the extracted text.
41	
42	Fires **at most once per task** and **only** when a tesseract call emitted the
43	low-resolution signal, so it is invisible to the (large) majority of non-OCR
44	tasks — zero added cost or regression surface there.  It mutates only the
45	tool-result string (the same shape as ``CustomEditToolProcessor``); it never
46	inserts, drops, or reorders messages.
47	"""
48	from __future__ import annotations
49	
50	import dataclasses
51	import re
52	
53	from harnessx.core.processor import MultiHookProcessor
54	
55	# --- detection --------------------------------------------------------------
56	
57	# The agent asked tesseract to read an image (CLI or the pytesseract binding).
58	_OCR_INVOKE_RE = re.compile(r"\btesseract\b|pytesseract|image_to_string", re.IGNORECASE)
59	
60	# Tesseract's own signal that the image lacks DPI metadata / is being read at a
61	# guessed (usually too-low) resolution — the exact case where upscale + --dpi
62	# pays off most.
63	_LOW_RES_SIGNAL_RE = re.compile(
64	    r"Invalid resolution|Estimating resolution|Using \d+ instead", re.IGNORECASE
65	)
66	
67	_ADVISORY = (
68	    "\n\n[OcrQualityAdvisor] This tesseract call reported a missing / guessed "
69	    "image resolution, which is the most common cause of garbled OCR on small "
70	    "images. Before trusting this text (or building any parser/detector/mapping "
71	    "on top of it), re-run OCR with the standard high-accuracy preprocessing you "
72	    "have not yet applied:\n"
73	    "  1. UPSCALE the image ~3-4x (e.g. PIL `img.resize((w*3, h*3), "
74	    "Image.LANCZOS)`) — this alone usually fixes small/low-DPI images.\n"
75	    "  2. Pass an EXPLICIT dpi to tesseract: `--dpi 300` (or `tesseract ... "
76	    "-c user_defined_dpi=300`).\n"
77	    "  3. Convert to grayscale and binarize (threshold) so glyph edges are "
78	    "crisp; try `--psm 6` for a uniform block of text.\n"
79	    "  4. Re-run and DIFF the new text against this one; treat characters that "
80	    "change between runs as low-confidence and disambiguate visually "
81	    "(common confusions: I/l/1, 0/O, 5/S, 8/B, rn/m, and dropped/added spaces).\n"
82	    "Verify the extracted text against any sample input/expected output the task "
83	    "provides before committing downstream code."
84	)
85	
86	
87	class OcrQualityAdvisor(MultiHookProcessor):
88	    """One-time nudge toward high-accuracy tesseract preprocessing on low-DPI OCR."""
89	
90	    _singleton_group = "ocr_quality_advisor"
91	    _order = 32  # after CustomEditToolProcessor(30), before CustomSelfVerifyProcessor(90)
92	
93	    def __init__(self) -> None:
94	        self._ocr_calls: set[str] = set()
95	        self._fired: bool = False
96	
97	    async def on_task_start(self, event):
98	        self._ocr_calls.clear()
99	        self._fired = False
100	        yield event
101	
102	    async def on_before_tool(self, event):
103	        if event.tool_name == "Bash":
104	            command = (event.tool_input or {}).get("command", "") or ""
105	            if _OCR_INVOKE_RE.search(command):
106	                self._ocr_calls.add(event.tool_call_id)
107	        yield event
108	
109	    async def on_after_tool(self, event):
110	        if self._fired:
111	            self._ocr_calls.discard(event.tool_call_id)
112	            yield event
113	            return
114	
115	        was_ocr = event.tool_call_id in self._ocr_calls
116	        self._ocr_calls.discard(event.tool_call_id)
117	
118	        result_text = event.result or ""
119	        if was_ocr and _LOW_RES_SIGNAL_RE.search(result_text):
120	            self._fired = True
121	            yield dataclasses.replace(event, result=result_text + _ADVISORY)
122	        else:
123	            yield event
124	
125	    async def on_task_end(self, event):
126	        self._ocr_calls.clear()
127	        self._fired = False
128	        yield event
