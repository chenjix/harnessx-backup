14	def _run(
15	    cmd: list[str],
16	    *,
17	    timeout: float | None = None,
18	    cwd: Path | None = None,
19	) -> subprocess.CompletedProcess[str]:
20	    return subprocess.run(
21	        cmd,
22	        text=True,
23	        capture_output=True,
24	        timeout=timeout,
25	        cwd=str(cwd) if cwd else None,
26	    )

29	def image_tag(task_id: str, container_def: str) -> str:
30	    h = hashlib.sha1(container_def.encode()).hexdigest()[:12]
31	    safe = task_id.replace("/", "_")
32	    return f"tmax-eval:{safe}-{h}"

35	_SHARED_BASE_OK: bool | None = None
36	
37	
38	def _shared_base_tag() -> str | None:
39	    """Tag of an image holding the preamble every taxonomy def repeats, if usable.
40	
41	    Set ``TMAX_SHARED_BASE_TAG`` (see scripts/tmax/prebuild_tmax_images.sh
42	    SHARED_BASE=1) to rebase task images onto it. Every taxonomy container_def is
43	    ``FROM ubuntu:22.04`` plus a %post that reinstalls python3/pip/pytest, and the
44	    whole %post becomes one layer — so without this each task image carries its
45	    own ~0.4G copy of the same interpreter. Rebasing keeps the per-task layer down
46	    to what the task itself adds, which is the difference between ~60G and ~15G
47	    for a 152-task set. The image TAG is unchanged either way.
48	    """
49	    global _SHARED_BASE_OK
50	    tag = (os.environ.get("TMAX_SHARED_BASE_TAG") or "").strip()
51	    if not tag:
52	        return None
53	    if _SHARED_BASE_OK is None:
54	        _SHARED_BASE_OK = _run(["docker", "image", "inspect", tag], timeout=30).returncode == 0
55	    return tag if _SHARED_BASE_OK else None
56	
57	
58	def _maybe_rebase(dockerfile: Path) -> None:
59	    """Point a materialized Dockerfile at the shared base, when there is one."""
60	    tag = _shared_base_tag()
61	    if tag is None:
62	        return
63	    text = dockerfile.read_text(encoding="utf-8")
64	    # Only ubuntu:22.04 defs — do not silently swap someone else's base out.
65	    if not text.startswith("FROM ubuntu:22.04\n"):
66	        return
67	    dockerfile.write_text(
68	        text.replace("FROM ubuntu:22.04\n", f"FROM {tag}\n", 1), encoding="utf-8"
69	    )

72	def build_image(
73	    task_id: str,
74	    container_def: str,
75	    work_root: Path,
76	    *,
77	    rebuild: bool = False,
78	    build_timeout: float = 1800,
79	) -> str:
80	    tag = image_tag(task_id, container_def)
81	    if not rebuild:
82	        probe = _run(["docker", "image", "inspect", tag], timeout=30)
83	        if probe.returncode == 0:
84	            return tag
85	    ctx = work_root / "build" / task_id
86	    ctx.mkdir(parents=True, exist_ok=True)
87	    for p in ctx.iterdir():
88	        if p.is_file():
89	            p.unlink()
90	    _maybe_rebase(materialize_build_context(container_def, ctx))
91	    proc = _run(
92	        ["docker", "build", "--network=host", "-t", tag, str(ctx)],
93	        timeout=build_timeout,
94	    )
95	    if proc.returncode != 0:
96	        raise RuntimeError(
97	            f"docker build failed for {task_id} (rc={proc.returncode}):\n"
98	            f"{proc.stdout[-2000:]}\n{proc.stderr[-4000:]}"
99	        )
100	    return tag

103	def start_container(task_id: str, image: str, *, network: str = "bridge") -> str:
104	    """Start a long-lived container. Default bridge so apt-using agent cmds can work."""
105	    name = f"tmax-{task_id.replace('_', '')[:20]}-{int(time.time())}"[:63]
106	    proc = _run(
107	        [
108	            "docker",
109	            "run",
110	            "-d",
111	            "--name",
112	            name,
113	            "--network",
114	            network,
115	            "--workdir",
116	            "/home/user",
117	            image,
118	            "sleep",
119	            "infinity",
120	        ],
121	        timeout=60,
122	    )
123	    if proc.returncode != 0:
124	        raise RuntimeError(
125	            f"docker run failed for {task_id}: {proc.stderr or proc.stdout}"
126	        )
127	    return (proc.stdout or "").strip() or name

130	def stop_container(container: str, *, remove: bool = True) -> None:
131	    _run(["docker", "kill", container], timeout=60)
132	    if remove:
133	        _run(["docker", "rm", "-f", container], timeout=60)

136	def exec_in(
137	    container: str,
138	    command: str,
139	    *,
140	    timeout: float = 120,
141	    workdir: str = "/home/user",
142	) -> tuple[int, str]:
143	    try:
144	        proc = _run(
145	            [
146	                "docker",
147	                "exec",
148	                "-w",
149	                workdir,
150	                container,
151	                "bash",
152	                "-lc",
153	                command,
154	            ],
155	            timeout=timeout + 5,
156	        )
157	    except subprocess.TimeoutExpired:
158	        return 124, f"Error: command timed out after {timeout}s"
159	    out = (proc.stdout or "") + (proc.stderr or "")
160	    return proc.returncode, out

163	def write_file(container: str, path: str, content: str) -> None:
164	    # Ensure parent dir exists, then stream bytes via stdin.
165	    parent = str(Path(path).parent)
166	    _run(
167	        ["docker", "exec", container, "bash", "-lc", f"mkdir -p {parent}"],
168	        timeout=30,
169	    )
170	    proc = subprocess.run(
171	        ["docker", "exec", "-i", container, "bash", "-lc", f"cat > {path}"],
172	        input=content,
173	        text=True,
174	        capture_output=True,
175	        timeout=60,
176	    )
177	    if proc.returncode != 0:
178	        raise RuntimeError(
179	            f"failed to write {path} in {container}: {proc.stderr or proc.stdout}"
180	        )

183	def run_pytest(container: str, test_src: str, *, dest: str) -> dict[str, Any]:
184	    write_file(container, dest, test_src)
185	    # Ensure pytest is available (most images install it in %post).
186	    rc, out = exec_in(
187	        container,
188	        f"python3 -m pytest -q {dest} 2>&1 || python3 -m pytest -q {dest} 2>&1",
189	        timeout=180,
190	    )
191	    # If pytest module missing, try installing once.
192	    if rc != 0 and "No module named pytest" in out:
193	        exec_in(container, "pip3 install -q pytest", timeout=180)
194	        rc, out = exec_in(container, f"python3 -m pytest -q {dest} 2>&1", timeout=180)
195	    return {"passed": rc == 0, "rc": rc, "output": out[-8000:]}
