54	_REDIRECT_WRITE_RE = re.compile(r"(?<![<>2&\d])>{1,2}\s*([^\s;&|><\n'\"]+)")
55	_SED_INPLACE_RE = re.compile(r"\bsed\s+(?:-[a-zA-Z]*i[a-zA-Z]*|-i\S*)\s+\S+\s+([^\s;&|><\n'\"]+)")
56	_TEE_WRITE_RE = re.compile(r"\btee\s+(?:-\S+\s+)*([^\s;&|><\n'\"]+)")
57	_SKIP_PATH_RE = re.compile(r"^(/dev/|/proc/|/sys/|-|\d+$)")
58	
59	
60	def _extract_written_files(cmd: str) -> list[str]:
61	    """Return deduplicated file paths that a bash command writes to."""
62	    if ">" not in cmd and "sed" not in cmd and "tee" not in cmd:
63	        return []
64	    paths: set[str] = set()
65	    for pattern in (_REDIRECT_WRITE_RE, _SED_INPLACE_RE, _TEE_WRITE_RE):
66	        for m in pattern.finditer(cmd):
67	            p = m.group(1).strip("'\"")
68	            if p and not _SKIP_PATH_RE.match(p):
69	                paths.add(p)
70	    return list(paths)

103	class CustomSelfVerifyProcessor(MultiHookProcessor):
104	    """Inject a one-shot verification prompt when the model tries to exit without tool calls.
105	
106	    Fires at most once per task run.  On the next no-tool-call turn it stays silent.
107	    """
108	
109	    _singleton_group = "tb2_self_verify"
110	    _order = 90
111	
112	    def __init__(self) -> None:
113	        self._verified = False
114	        self._pending_message: str = ""
115	
116	    async def on_task_start(self, event: TaskStartEvent):
117	        self._verified = False
118	        self._pending_message = ""
119	        yield event
120	
121	    async def on_before_model(self, event: BeforeModelEvent):
122	        if not self._pending_message:
123	            yield event
124	            return
125	        msg = self._pending_message
126	        self._pending_message = ""
127	        # last message is a tool result (role≠user) → append exactly +1 user
128	        yield dataclasses.replace(
129	            event,
130	            messages=event.messages + (Message(role="user", content=msg),),
131	        )
132	
133	    async def on_after_model(self, event: ModelResponseEvent):
134	        exit_intent = event.finish_reason in ("end_turn", "stop") and not event.tool_calls
135	        if exit_intent and not self._verified:
136	            self._verified = True
137	            self._pending_message = _SELF_VERIFY_MSG
138	            keepalive = ToolCall(
139	                id=f"sv-{uuid.uuid4().hex[:8]}",
140	                name=_SELF_VERIFY_TOOL,
141	                input={},
142	            )
143	            yield dataclasses.replace(event, tool_calls=(keepalive,))
144	        else:
145	            yield event
146	
147	    async def on_before_tool(self, event: ToolCallEvent):
148	        if event.tool_name == _SELF_VERIFY_TOOL:
149	            yield dataclasses.replace(event, approved=False, synthetic_result=_SELF_VERIFY_ACK)
150	        else:
151	            yield event
152	
153	    async def on_task_end(self, event: TaskEndEvent):
154	        self._verified = False
155	        self._pending_message = ""
156	        yield event

159	class CustomEditToolProcessor(MultiHookProcessor):
160	    """Detect repeated Bash-based file edits and nudge the model to try a different approach.
161	
162	    TB2 only exposes Bash, so write operations are detected via command patterns
163	    (redirects, sed -i, tee).  When the same file is edited more than
164	    ``threshold`` times, a warning is appended to the tool result and the
165	    per-file counter is reset.
166	    """
167	
168	    _singleton_group = "bash_edit_detector"
169	    _order = 30
170	
171	    def __init__(self, threshold: int = 7) -> None:
172	        self.threshold = threshold
173	        self._edit_counts: dict[str, int] = {}
174	        self._pending: dict[str, list[str]] = {}
175	
176	    async def on_task_start(self, event: TaskStartEvent):
177	        self._edit_counts.clear()
178	        self._pending.clear()
179	        yield event
180	
181	    async def on_before_tool(self, event: ToolCallEvent):
182	        if event.tool_name == "Bash":
183	            paths = _extract_written_files(event.tool_input.get("command", ""))
184	            if paths:
185	                self._pending[event.tool_call_id] = paths
186	        yield event
187	
188	    async def on_after_tool(self, event: ToolResultEvent):
189	        paths = self._pending.pop(event.tool_call_id, [])
190	        injections: list[str] = []
191	        for path in paths:
192	            count = self._edit_counts.get(path, 0) + 1
193	            self._edit_counts[path] = count
194	            if count > self.threshold:
195	                injections.append(_EDIT_LIMIT_WARN.format(path=path, threshold=self.threshold))
196	                self._edit_counts[path] = 0
197	        if injections:
198	            yield dataclasses.replace(event, result=(event.result or "") + "".join(injections))
199	        else:
200	            yield event
201	
202	    async def on_task_end(self, event: TaskEndEvent):
203	        self._edit_counts.clear()
204	        self._pending.clear()
205	        yield event

212	_TASK_TIME_WARN_70 = (
213	    "[TaskTimeReminder] You have used 70% of your time budget ({elapsed:.0f}s / {total:.0f}s). "
214	    "Stop exploring. Focus exclusively on implementing and writing all required output files. "
215	    "Run `ls -lh` on each expected output path to confirm they exist."
216	)
217	_TASK_TIME_WARN_90 = (
218	    "[TaskTimeReminder] CRITICAL: You have used 90% of your time budget ({elapsed:.0f}s / {total:.0f}s, "
219	    "{remaining:.0f}s left). "
220	    "If you have NOT yet written all required output files, do so IMMEDIATELY — "
221	    "the task will be terminated soon and any missing output files will cause an automatic score of 0. "
222	    "Do NOT start any new implementation. Write and verify output files NOW."

226	class TaskTimeReminderProcessor(MultiHookProcessor):
227	    _singleton_group = "task_time_reminder"
228	    _order = 6
229	
230	    def __init__(
231	        self,
232	        timeout_seconds: float | None = None,
233	        warn_at: tuple[float, ...] = (0.70, 0.90),
234	    ) -> None:
235	        self._timeout = timeout_seconds
236	        self._warn_at = sorted(warn_at)
237	        self._triggered: set[float] = set()
238	        self._start: float = 0.0
239	
240	    async def on_task_start(self, event: TaskStartEvent):
241	        self._triggered = set()
242	        self._start = time.monotonic()
243	        yield event
244	
245	    async def on_step_start(self, event: StepStartEvent):
246	        if not self._timeout or self._timeout <= 0:
247	            yield event
248	            return
249	
250	        elapsed = time.monotonic() - self._start
251	        pct = elapsed / self._timeout
252	        remaining = self._timeout - elapsed
253	
254	        warn: str | None = None
255	        for threshold in reversed(self._warn_at):
256	            if pct >= threshold and threshold not in self._triggered:
257	                self._triggered.add(threshold)
258	                if threshold >= 0.90:
259	                    warn = _TASK_TIME_WARN_90.format(elapsed=elapsed, total=self._timeout, remaining=remaining)
260	                else:
261	                    warn = _TASK_TIME_WARN_70.format(elapsed=elapsed, total=self._timeout)
262	                break
263	
264	        if warn:
265	            msg = Message(role="user", content=warn)
266	            yield dataclasses.replace(
267	                event,
268	                messages=event.messages + (msg,),
269	                raw_messages=event.raw_messages + (msg,),
270	            )
271	        else:
272	            yield event
273	
274	    async def on_task_end(self, event: TaskEndEvent):
275	        self._triggered = set()
276	        yield event

284	class PostCompactionRefreshProcessor(MultiHookProcessor):
285	    """After context compaction, re-inject a fresh workspace snapshot from the
286	    sandbox so the model knows what files currently exist rather than relying on
287	    compressed-away history.
288	
289	    Detects compaction by a significant drop in message count between steps
290	    (same heuristic used by LoopDetectionProcessor).
291	    """
292	
293	    _singleton_group = "tb2_post_compaction_refresh"
294	    _order = 11  # runs after CompactionProcessor (8)
295	
296	    def __init__(self, drop_threshold: int = 5) -> None:
297	        self._drop_threshold = drop_threshold
298	        self._prev_count: int = 0
299	
300	    async def on_task_start(self, event: TaskStartEvent):
301	        self._prev_count = 0
302	        yield event
303	
304	    async def on_step_start(self, event: StepStartEvent):
305	        current_count = len(event.messages)
306	        prev = self._prev_count
307	        self._prev_count = current_count
308	
309	        if not (prev > 0 and (prev - current_count) >= self._drop_threshold):
310	            yield event
311	            return
312	
313	        from harnessx.sandbox.base import get_current_sandbox
314	
315	        sandbox = get_current_sandbox()
316	        if sandbox is None:
317	            yield event
318	            return
319	
320	        try:
321	            ls_out = await sandbox.exec("ls -la /app 2>/dev/null | head -30", timeout=8)
322	        except Exception:
323	            yield event
324	            return
325	
326	        procs_out = ""
327	        try:
328	            procs_out = await sandbox.exec(
329	                "ps aux --no-headers 2>/dev/null | grep -v '\\[' "
330	                "| awk '{print $11}' | sort -u | grep -v '^ps$' | head -8",
331	                timeout=5,
332	            )
333	        except Exception:
334	            pass
335	
336	        snapshot = f"[PostCompaction] Context was compressed. Current workspace state:\n```\n{ls_out.strip()}\n```"
337	        if procs_out and procs_out.strip():
338	            snapshot += f"\nRunning processes: {procs_out.strip()}"
339	
340	        msg = Message(role="user", content=snapshot)
341	        yield dataclasses.replace(
342	            event,
343	            messages=event.messages + (msg,),
344	            raw_messages=event.raw_messages + (msg,),
345	        )
346	
347	    async def on_task_end(self, event: TaskEndEvent):
348	        self._prev_count = 0
349	        yield event

356	_TB2_SYSTEM = """\
357	# System
358	You are an expert software engineer solving terminal-based programming tasks in a Linux sandbox.
359	
360	## Key Facts About This Environment
361	- You are inside an isolated Docker container.
362	- Your job is to implement the task correctly so the external verifier passes when it runs.
363	
364	## Workflow
365	1. **Read the task carefully.** Before writing any code, understand:
366	   - Required output format, file paths, field names, CLI flags
367	   - Accuracy or size constraints
368	   - What a correct solution looks like from the outside
369	2. **Explore the environment first.**
370	```bash
371	   ls /app && find /app -name '*.md' -o -name '*.txt' -o -name '*.sh' 2>/dev/null | head -20
372	```
373	3. **For complex tasks, decompose before implementing.** Break the problem into concrete subtasks. Solve and verify each subtask before moving to the next — do not attempt everything at once.
374	4. **Implement the solution.** Match the task's specified interface exactly — file paths, output format, and CLI behavior all matter to the verifier.
375	5. **Write outputs to the exact paths the task specifies.** Re-read the task description to confirm paths before writing anything.
376	6. **Verify outputs explicitly** — do not assume a file was written just because a script ran without errors. Check with `ls` and inspect the contents:
377	```bash
378	   ls -lh /path/to/required/output   # confirm file exists and is non-empty
379	   head -20 /path/to/required/output  # confirm content is correct
380	```
381	7. **For background services**, keep them running after you exit and verify they are reachable:
382	```bash
383	   nohup python server.py > /tmp/server.log 2>&1 &
384	   sleep 1 && curl -sf http://localhost:8080/health
385	```
386	8. **Do not use unbounded background commands.** If a command's runtime is unpredictable, run it synchronously. Only use `&` / `nohup` for long-running services the verifier will connect to.
387	9. **Do not run `apt-get update`.** Package lists are pre-cached — `apt-get install -y <pkg>` works directly and completes in seconds.
388	
389	## Definition of Done
390	- All required output files exist at the exact paths specified in the task — confirmed with `ls`
391	- Any required services are running and reachable
392	- You have inspected the actual output values and confirmed they are correct
393	- No temporary or debug files remain that could interfere with the verifier
394	
395	When all checks pass, end your final message with:
396	**SUCCESS: task complete. Output files confirmed at [list each required path].**
397	"""

414	def make_tb2_harness_config(
415	    *,
416	    timeout_seconds: int | None = None,
417	    workspace_path: str = WORKSPACE_PATH,
418	    output_limit: int = OUTPUT_LIMIT,
419	) -> HarnessConfig:
420	    """Build a serializable ``HarnessConfig`` for Terminal Bench 2.0.
421	
422	    This config contains only the processor pipeline and tool registry — no
423	    runtime-only slots (``sandbox_provider``, ``tracer``).  It can be saved to
424	    YAML via ``cfg.to_yaml_file(path)`` and evolved by the meta-harness.
425	
426	    The agent injects the runtime slots at run time::
427	
428	        base = make_tb2_harness_config(timeout_seconds=task_timeout)
429	        harness_config = base.copy(
430	            sandbox_provider=HarborSandboxProvider(environment, ...),
431	            tracer=HarnessJournal(...),
432	        )
433	        harness = ModelConfig(main=provider).agentic(harness_config)
434	
435	    Args:
436	        timeout_seconds:  Task timeout disclosed to the model via
437	                          ``EnvironmentContextInjector`` and
438	                          ``TaskTimeReminderProcessor``.
439	        workspace_path:   Absolute path inside the container where task files live.
440	        output_limit:     Maximum characters captured per tool call (passed
441	                          through to ``HarborSandboxProvider`` at run time via
442	                          the agent; stored here for reference only).
443	    """
444	    _ = output_limit  # documented here; consumed by HarborSandboxProvider in agent.py
445	
446	    return (
447	        HarnessBuilder()
448	        .add(SystemPromptProcessor(_StaticSystemPromptBuilder()), order=1)
449	        .add(
450	            EnvironmentContextInjector(
451	                working_dir=workspace_path,
452	                timeout_seconds=timeout_seconds,
453	                show_project_dir=False,
454	            ),
455	            order=2,
456	        )
457	        .add(TaskTimeReminderProcessor(timeout_seconds=timeout_seconds))
458	        .add(CompactionProcessor())
459	        .add(PostCompactionRefreshProcessor())
460	        .add(ToolCallCorrectionLayer())
461	        .add(ParseRetryProcessor())
462	        .add(BgInstallGuard())
463	        .add(CustomEditToolProcessor())
464	        .add(CustomSelfVerifyProcessor())
465	        .slot(tool_registry=_build_tb2_tools())
466	    ).build()
