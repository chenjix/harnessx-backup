3	"""ServiceDepsReminderProcessor.
4	
5	Closes a structural failure mode observed on network-service tasks in the
6	tmax / terminal-bench-2 evaluation: the external verifier probes the agent's
7	HTTP/socket service using the Python ``requests`` client, but ``requests`` is
8	not guaranteed to be present in every task container. When it is absent the
9	verifier's ``test_final_state.py`` fails at *import/collection* time
10	(``ModuleNotFoundError: No module named 'requests'``) — a hard score of 0 even
11	when the agent's service is fully correct and reachable.
12	
13	The agent has one tool (``Bash``) and no visibility into the verifier files
14	(they are injected after the agent exits), so it cannot know the verifier's
15	Python dependencies. This processor supplies that missing context as a
16	*mechanical, one-shot* nudge: when the agent has demonstrably been building a
17	network service (bound a socket / started nginx / listened on a port / used an
18	HTTP server library / probed a local port) and then tries to finish, it is
19	reminded to make the common HTTP client library importable, installing it if
20	missing. The install command is idempotent and harmless if the library is
21	already present or if installation is not possible.
22	
23	This is a *class* fix, not a task fix: it fires on any task that exhibits the
24	network-service shape, keyed off the agent's own Bash activity, never off task
25	identifiers.
26	"""
27	
28	from __future__ import annotations
29	
30	import dataclasses
31	import re
32	import uuid
33	
34	from harnessx.core.events import (
35	    BeforeModelEvent,
36	    Message,
37	    ModelResponseEvent,
38	    TaskEndEvent,
39	    TaskStartEvent,
40	    ToolCall,
41	    ToolCallEvent,
42	    ToolResultEvent,
43	)
44	from harnessx.core.processor import MultiHookProcessor
45	
46	
47	# Signals in the agent's Bash commands that indicate it is standing up (or
48	# probing) a network / HTTP service whose verifier is likely to use an HTTP
49	# client library. Intentionally broad — false positives only cost one cheap,
50	# idempotent import check; false negatives re-introduce the silent verifier
51	# crash.
52	_SERVICE_SIGNAL_RE = re.compile(
53	    r"""
54	    (?:\bnginx\b)                       # reverse proxy / web server
55	    | (?:\blisten\s*\()                 # C/C++ socket listen()
56	    | (?:sockaddr_un|AF_UNIX|AF_INET)   # raw socket servers
57	    | (?:proxy_pass)                    # nginx upstream config
58	    | (?:httplib|cpp-httplib)           # common header-only C++ HTTP lib
59	    | (?:http\.server|BaseHTTPServer)   # python stdlib http servers
60	    | (?:flask|gunicorn|uvicorn|fastapi)# python web frameworks
61	    | (?:\bgo\s+run\b.*http)            # go http servers
62	    | (?:127\.0\.0\.1:\d+)              # binding/probing a local port
63	    | (?:localhost:\d+)
64	    | (?:/dev/tcp/)                     # bash raw HTTP probe (curl/wget absent)
65	    | (?:\bcurl\b|\bwget\b).*http       # http probe via client
66	    | (?:\.sock\b)                      # unix-domain socket file
67	    """,
68	    re.IGNORECASE | re.VERBOSE,
69	)
70	
71	_DEPS_TOOL = "_svc_deps_reminder"
72	_DEPS_ACK = "Dependency check acknowledged. See the message above."
73	
74	_DEPS_MSG = """\
75	[VerifierDependencyCheck] This task builds or exposes a network/HTTP service. \
76	Automated verifiers for service tasks commonly probe the endpoint from Python \
77	using the `requests` library, and they import it at module load — so if \
78	`requests` is not importable in this environment, the verification crashes at \
79	collection time and the task scores 0 regardless of whether your service works.
80	
81	Before you finish, make the standard Python HTTP client importable. Run an \
82	idempotent check-then-install (safe to run even if it is already present, and \
83	harmless if installation is not possible):
84	
85	```bash
86	python3 -c 'import requests' 2>/dev/null \\
87	  || pip install --quiet requests 2>/dev/null \\
88	  || pip3 install --quiet requests 2>/dev/null || true
89	python3 -c 'import requests, sys; print("requests OK", requests.__version__)' \\
90	  || echo "requests still unavailable"
91	```
92	
93	This does not replace verifying your own service — keep confirming it responds \
94	correctly. It only ensures the verifier's own imports can succeed."""
95	
96	
97	class ServiceDepsReminderProcessor(MultiHookProcessor):
98	    """One-shot reminder to ensure verifier HTTP-client deps are importable.
99	
100	    Fires at most once per task, and only when the agent's Bash activity has
101	    matched a network-service signal. Coordinates with any other exit-intent
102	    processor (e.g. the self-verify checklist) by only acting on a turn that
103	    still has no tool calls: if another processor has already converted the
104	    current exit attempt into a keepalive tool call, this processor stays
105	    silent and fires on the next genuine exit attempt instead.
106	    """
107	
108	    _singleton_group = "svc_deps_reminder"
109	    # Run after CustomSelfVerifyProcessor (_order=90) so the two exit-intent
110	    # hooks serialize rather than both rewriting tool_calls on the same turn.
111	    _order = 91
112	
113	    def __init__(self) -> None:
114	        self._service_seen = False
115	        self._reminded = False
116	        self._pending_message = ""
117	
118	    async def on_task_start(self, event: TaskStartEvent):
119	        self._service_seen = False
120	        self._reminded = False
121	        self._pending_message = ""
122	        yield event
123	
124	    async def on_before_tool(self, event: ToolCallEvent):
125	        # Absorb our own keepalive tool call so it never reaches the sandbox.
126	        if event.tool_name == _DEPS_TOOL:
127	            yield dataclasses.replace(
128	                event, approved=False, synthetic_result=_DEPS_ACK
129	            )
130	            return
131	        if event.tool_name == "Bash":
132	            cmd = event.tool_input.get("command", "") or ""
133	            if _SERVICE_SIGNAL_RE.search(cmd):
134	                self._service_seen = True
135	        yield event
136	
137	    async def on_before_model(self, event: BeforeModelEvent):
138	        if not self._pending_message:
139	            yield event
140	            return
141	        msg = self._pending_message
142	        self._pending_message = ""
143	        yield dataclasses.replace(
144	            event,
145	            messages=event.messages + (Message(role="user", content=msg),),
146	        )
147	
148	    async def on_after_model(self, event: ModelResponseEvent):
149	        exit_intent = (
150	            event.finish_reason in ("end_turn", "stop") and not event.tool_calls
151	        )
152	        if exit_intent and self._service_seen and not self._reminded:
153	            self._reminded = True
154	            self._pending_message = _DEPS_MSG
155	            keepalive = ToolCall(
156	                id=f"svc-{uuid.uuid4().hex[:8]}",
157	                name=_DEPS_TOOL,
158	                input={},
159	            )
160	            yield dataclasses.replace(event, tool_calls=(keepalive,))
161	        else:
162	            yield event
163	
164	    async def on_task_end(self, event: TaskEndEvent):
165	        self._service_seen = False
166	        self._reminded = False
167	        self._pending_message = ""
168	        yield event
