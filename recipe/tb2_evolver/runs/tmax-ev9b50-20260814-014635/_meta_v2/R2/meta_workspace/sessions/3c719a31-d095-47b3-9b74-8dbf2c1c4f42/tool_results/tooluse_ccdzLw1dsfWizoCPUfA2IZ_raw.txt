84	def _resolve_system_prompt(
85	    harness_config: Path | None,
86	    system_prompt_file: Path | None,
87	) -> str | None:
88	    """Resolve Tmax agent system prompt for an evolve round.
89	
90	    Preference order:
91	      1. --system-prompt-file
92	      2. system_prompt.txt next to harness config (MetaAgent can write this)
93	      3. templates/*.j2 under the harness config directory
94	      4. None → agent_loop default
95	    """
96	    if system_prompt_file is not None and system_prompt_file.is_file():
97	        return system_prompt_file.read_text(encoding="utf-8")
98	    if harness_config is not None:
99	        cfg = Path(harness_config).resolve()
100	        sibling = cfg.parent / "system_prompt.txt"
101	        if sibling.is_file():
102	            return sibling.read_text(encoding="utf-8")
103	        templates = cfg.parent / "templates"
104	        if templates.is_dir():
105	            j2s = sorted(templates.glob("*.j2"))
106	            if j2s:
107	                return j2s[0].read_text(encoding="utf-8")
108	    env_prompt = (os.environ.get("TMAX_SYSTEM_PROMPT_FILE") or "").strip()
109	    if env_prompt and Path(env_prompt).is_file():
110	        return Path(env_prompt).read_text(encoding="utf-8")
111	    return None
