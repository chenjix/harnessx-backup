#!/usr/bin/env python3
"""
Deployment Monitor Daemon

Monitors disk usage in /home/user/logs/ and controls worker processes
to prevent disk quota exhaustion during deployment.
"""

import os
import time
import signal
import subprocess
import glob
import sys

LOGS_DIR = "/home/user/logs"
LOG_PATTERN = os.path.join(LOGS_DIR, "*.log")
THRESHOLD_BYTES = 40 * 1024 * 1024  # 40 MB
SLEEP_INTERVAL = 0.1


def get_total_logs_size():
    """Calculate total size of all .log files in the logs directory."""
    total_size = 0
    try:
        for filepath in glob.glob(LOG_PATTERN):
            if os.path.isfile(filepath):
                try:
                    total_size += os.path.getsize(filepath)
                except OSError:
                    pass
    except OSError:
        pass
    return total_size


def get_worker_processes():
    """Get PIDs of all running worker_sim.py processes."""
    worker_pids = []
    try:
        result = subprocess.run(
            ["pgrep", "-f", "worker_sim.py"],
            capture_output=True,
            text=True,
            timeout=2
        )
        if result.returncode == 0:
            for line in result.stdout.strip().split('\n'):
                line = line.strip()
                if line:
                    try:
                        pid = int(line)
                        # Verify the process is actually running
                        os.kill(pid, 0)
                        worker_pids.append(pid)
                    except (ValueError, OSError):
                        pass
    except (subprocess.TimeoutExpired, ValueError, FileNotFoundError, OSError):
        pass
    return worker_pids


def send_signal_to_workers(signum):
    """Send a signal to all worker processes."""
    worker_pids = get_worker_processes()
    for pid in worker_pids:
        try:
            os.kill(pid, signum)
        except (OSError, ProcessLookupError):
            pass


def truncate_log_files():
    """Truncate all .log files in the logs directory to 0 bytes."""
    for filepath in glob.glob(LOG_PATTERN):
        try:
            with open(filepath, 'r+') as f:
                f.seek(0, 2)  # Go to end
                f.seek(0)     # Go to beginning
                f.truncate()
        except (OSError, IOError):
            pass
